# 전역 이벤트 루프 및 Redis Client 도입

## 1. 설계 원칙

- 단일 이벤트 루프: Celery Worker 프로세스당 하나의 공유 이벤트 루프
- 연결 풀링: Redis 클라이언트 연결 풀을 통한 효율적 연결 관리
- Graceful Shutdown: 워커 종료 시 리소스 정리 보장

## 2. 구현 상세

### 2.1 전역 이벤트 루프 관리자

신규 파일: apps/shared/async_runtime.py

```python
"""
전역 AsyncIO 런타임 관리

Celery Worker 프로세스당 하나의 이벤트 루프를 공유합니다.
- 이벤트 루프 생성/종료 오버헤드 제거
- 연결 풀 등 비동기 리소스의 수명 관리 단순화
"""

import asyncio
import threading
from typing import Optional
import atexit

_loop: Optional[asyncio.AbstractEventLoop] = None
_loop_thread: Optional[threading.Thread] = None
_lock = threading.Lock()


def get_event_loop() -> asyncio.AbstractEventLoop:
    """
    전역 이벤트 루프 반환 (Lazy 초기화)

    스레드 세이프하게 이벤트 루프를 생성하고 백그라운드 스레드에서 실행합니다.
    """
    global _loop, _loop_thread

    with _lock:
        if _loop is None or _loop.is_closed():
            _loop = asyncio.new_event_loop()
            _loop_thread = threading.Thread(
                target=_loop.run_forever,
                daemon=True,
                name="async-runtime"
            )
            _loop_thread.start()
            atexit.register(shutdown)

    return _loop


def run_coroutine(coro):
    """
    코루틴을 전역 이벤트 루프에서 실행하고 결과 반환

    동기 컨텍스트(Celery 태스크)에서 비동기 코드를 실행할 때 사용합니다.
    """
    loop = get_event_loop()
    future = asyncio.run_coroutine_threadsafe(coro, loop)
    return future.result()


def shutdown():
    """전역 이벤트 루프 종료"""
    global _loop, _loop_thread

    with _lock:
        if _loop and _loop.is_running():
            _loop.call_soon_threadsafe(_loop.stop)
        if _loop_thread:
            _loop_thread.join(timeout=5.0)
        if _loop and not _loop.is_closed():
            _loop.close()
        _loop = None
        _loop_thread = None
```

### 2.2 전역 Redis Client 풀

신규 파일: apps/shared/redis_pool.py

```python
"""
전역 Redis 연결 풀 관리

모든 비동기 Redis 작업은 이 풀을 통해 수행됩니다.
"""

import redis.asyncio as aioredis
from typing import Optional
import os

_redis_pool: Optional[aioredis.ConnectionPool] = None
_redis_client: Optional[aioredis.Redis] = None


async def get_redis_client() -> aioredis.Redis:
    """전역 Redis 클라이언트 반환 (연결 풀 사용)"""
    global _redis_pool, _redis_client

    if _redis_client is None:
        redis_url = os.getenv("REDIS_URL", "redis://localhost:6379/0")

        _redis_pool = aioredis.ConnectionPool.from_url(
            redis_url,
            max_connections=50,
            decode_responses=True,
        )
        _redis_client = aioredis.Redis(connection_pool=_redis_pool)

    return _redis_client


async def close_redis_pool():
    """연결 풀 종료"""
    global _redis_pool, _redis_client

    if _redis_client:
        await _redis_client.close()
    if _redis_pool:
        await _redis_pool.disconnect()

    _redis_client = None
    _redis_pool = None
```

## 3. Celery Worker 시그널 연동

수정 파일: apps/shared/celery_app.py

```python
from celery.signals import worker_shutdown

@worker_shutdown.connect
def cleanup_on_shutdown(sender, **kwargs):
    """Worker 종료 시 전역 리소스 정리"""
    from apps.shared.async_runtime import shutdown, run_coroutine
    from apps.shared.redis_pool import close_redis_pool

    try:
        run_coroutine(close_redis_pool())
    finally:
        shutdown()
```
