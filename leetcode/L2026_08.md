### [125. Valid Palindrome](https://leetcode.com/problems/valid-palindrome/)

```java
class Solution {
    public boolean isPalindrome(String s) {
        int n = s.length();

        StringBuilder sb = new StringBuilder();
        for (int idx = 0; idx < n; idx++) {
            char tmp = s.charAt(idx);
            if (Character.isLetterOrDigit(tmp)) {
                sb.append(Character.toLowerCase(tmp));
            }
        }

        String t1 = sb.toString();
        String t2 = sb.reverse().toString();

        return t1.equals(t2);
    }
}
```

### [125. Valid Palindrome](https://leetcode.com/problems/valid-palindrome/)

```java
class Solution {
    public boolean isPalindrome(String s) {
        int pLeft = 0;
        int pRight = s.length() - 1;

        while (pLeft < pRight) {
            while (pLeft < s.length() && !Character.isLetterOrDigit(s.charAt(pLeft))) {
                pLeft++;
            }
            while (0 <= pRight && !Character.isLetterOrDigit(s.charAt(pRight))) {
                pRight--;
            }
            if (pLeft >= pRight) return true;
            if (Character.toLowerCase(s.charAt(pLeft)) == Character.toLowerCase(s.charAt(pRight))) {
                pLeft++;
                pRight--;
            }
            else return false;
        }

        return true;
    }
}
```

### [392. Is Subsequence](https://leetcode.com/problems/is-subsequence/)

```java
class Solution {
    public boolean isSubsequence(String s, String t) {
        int ns = s.length();
        int nt = t.length();
        int ps = 0;
        int pt = 0;

        while (ps < ns) {
            while (pt < nt && t.charAt(pt) != s.charAt(ps)) {
                pt++;
            }
            if (pt == nt) return false;
            ps++;
            pt++;
        }

        if (ps == ns) return true;
        return false;
    }
}
```
