```java
class Solution {
    public int lengthOfLongestSubstring(String s) {
        int n = s.length();
        if (n == 0) return 0;

        HashMap<Character, Boolean> includeChecker = new HashMap<>();
        includeChecker.put(s.charAt(0), true);

        int rvLength = 1;
        int idxL = 0;
        int idxR = 0;
        while (idxL <= idxR) {
            if (idxL >= n || idxR >= n) break;
            if (idxR + 1 == n) break;

            char nextChar = s.charAt(idxR + 1);
            boolean isNextCharIncluded = includeChecker.getOrDefault(nextChar, false);
            if (!isNextCharIncluded) {
                if (++idxR >= n) break;
                includeChecker.put(nextChar, true);

                if (idxR - idxL + 1 > rvLength) {
                    rvLength = idxR - idxL + 1;
                }
            }
            else {
                char removingChar = s.charAt(idxL);
                if (idxL == idxR) {
                    idxL++;
                    includeChecker.put(removingChar, false);
                    idxR++;
                    includeChecker.put(s.charAt(idxR), true);
                }
                else {
                    idxL++;
                    includeChecker.put(removingChar, false);
                }
            }
        }

        return rvLength;
    }
```

```java
class Solution {
    public List<Integer> spiralOrder(int[][] matrix) {
        int borderLeft = 0;
        int borderRight = matrix[0].length - 1;
        int borderTop = 0;
        int borderBottom = matrix.length - 1;

        int[][] step = new int[4][2];
        step[0][0] = 0; step[0][1] = 1;
        step[1][0] = 1; step[1][1] = 0;
        step[2][0] = 0; step[2][1] = -1;
        step[3][0] = -1; step[3][1] = 0;

        List<Integer> rv = new ArrayList<>();
        int direction = 0;
        int r = 0;
        int c = 0;
        rv.add(matrix[r][c]);
        while (borderTop <= borderBottom && borderLeft <= borderRight) {
            while (true) {
                int nextR = r + step[direction][0];
                int nextC = c + step[direction][1];
                if (borderTop <= nextR && nextR <= borderBottom && borderLeft <= nextC && nextC <= borderRight) {
                    r = nextR;
                    c = nextC;
                    rv.add(matrix[r][c]);
                }
                else break;
            }
            if (direction == 0) borderTop++;
            else if (direction == 1) borderRight--;
            else if (direction == 2) borderBottom--;
            else if (direction == 3) borderLeft++;

            direction = (direction + 1) % 4;
        }
        return rv;
    }
}
```

```java
            matrix[base + sz - 1 - offset][base] = t3;
            matrix[base][base + offset] = t4;
        }
    }

        .
        .
        .
        (n - 1) ... (n - 1)
         */

            processFourLines(matrix, sz);
        }

    }
```

### [289. Game of Life](https://leetcode.com/problems/game-of-life/)

```java
class Solution {
    private int[][] updatingBoard;

    private int getNumberOfLiveFriends(int[][] board, int r, int c) {
        int m = board.length;
        int n = board[0].length;

        int rv = 0;
        if (0 <= r - 1 && r - 1 < m && board[r - 1][c] == 1) rv++;
        if (0 <= r + 1 && r + 1 < m && board[r + 1][c] == 1) rv++;
        if (0 <= c - 1 && c - 1 < n && board[r][c - 1] == 1) rv++;
        if (0 <= c + 1 && c + 1 < n && board[r][c + 1] == 1) rv++;
        if (0 <= r - 1 && r - 1 < m && 0 <= c - 1 && c - 1 < n && board[r - 1][c - 1] == 1) rv++;
        if (0 <= r - 1 && r - 1 < m && 0 <= c + 1 && c + 1 < n && board[r - 1][c + 1] == 1) rv++;
        if (0 <= r + 1 && r + 1 < m && 0 <= c - 1 && c - 1 < n && board[r + 1][c - 1] == 1) rv++;
        if (0 <= r + 1 && r + 1 < m && 0 <= c + 1 && c + 1 < n && board[r + 1][c + 1] == 1) rv++;

        return rv;
    }

    private void proceedOneGeneration(int[][] board) {
        int m = board.length;
        int n = board[0].length;

        for (int r = 0; r < m; r++) {
            for (int c = 0; c < n; c++) {
                int cnt = getNumberOfLiveFriends(board, r, c);
                if (cnt == 2) updatingBoard[r][c] = board[r][c];
                else if (cnt == 3) updatingBoard[r][c] = 1;
                else updatingBoard[r][c] = 0;
            }
        }

        for (int r = 0; r < m; r++) {
            for (int c = 0; c < n; c++) {
                board[r][c] = updatingBoard[r][c];
            }
        }

        return;
    }

    public void gameOfLife(int[][] board) {
        // 2 live neighbors: live
        // 3 live neighbors: live + reproduction
        int m = board.length;
        int n = board[0].length;

        updatingBoard = new int[m][n];
        proceedOneGeneration(board);

        return;
    }
}
```

### [290. Word Pattern](https://leetcode.com/problems/word-pattern/)

```java
class Solution {
    private int findOpeningIdx(String target, int idx) {
        while (target.charAt(idx) == ' ') {
            idx++;
            if (target.length() <= idx) {
                return -1;
            }
        }
        return idx;
    }

    private int findClosingIdx(String target, int idx) {
        while (target.charAt(idx) != ' ') {
            idx++;
            if (target.length() == idx) {
                return idx;
            }
        }
        return idx;
    }

    private String getWord(String target, int idx) {
        int openingIdx = findOpeningIdx(target, idx);
        int closingIdx = findClosingIdx(target, openingIdx);

        return target.substring(openingIdx, closingIdx);
    }

    public boolean wordPattern(String pattern, String target) {
        Map<Character, String> ptMap = new HashMap<>();
        Map<String, Character> tpMap = new HashMap<>();

        int idxPattern = 0;
        int idxTarget = 0;
        while (idxPattern < pattern.length() && idxTarget < target.length()) {
            int nextIdxTarget = findClosingIdx(target, findOpeningIdx(target, idxTarget));
            char p = pattern.charAt(idxPattern++);
            String w = getWord(target, idxTarget);

            if (ptMap.containsKey(p) && !ptMap.get(p).equals(w)) {
                return false;
            }
            else if (!ptMap.containsKey(p)) {
                ptMap.put(p, w);
            }

            if (tpMap.containsKey(w) && !tpMap.get(w).equals(p)) {
                return false;
            }
            else if (!tpMap.containsKey(w)) {
                tpMap.put(w, p);
            }

            idxTarget = nextIdxTarget;
        }

        if (idxPattern < pattern.length() || idxTarget < target.length()) {
            return false;
        }
        return true;
    }
}
```

```java
class Solution {
    private boolean isMatch(Stack<Character> pstack, char c) {
        if (pstack.isEmpty()) return false;

        char p = pstack.peek();
        if (p == '(') {
            if (c == ')') return true;
            return false;
        }
        else if (p == '{') {
            if (c == '}') return true;
            return false;
        }
        else if (p == '[') {
            if (c == ']') return true;
            return false;
        }

        return false;
    }

    public boolean isValid(String s) {
        int n = s.length();
        Stack<Character> pstack = new Stack<Character>();

        for (int idx = 0; idx < n; idx++) {
            char c = s.charAt(idx);
            if (c == '(' || c == '{' || c == '[') {
                pstack.push(c);
            }
            else if (c == ')' || c == '}' || c == ']') {
                if (isMatch(pstack, c)) {
                    pstack.pop();
                }
                else return false;
            }
        }

        return pstack.isEmpty();
    }
```

```java
class Solution {
    public String simplifyPath(String path) {
        String[] p = path.split("/");
        int n = p.length;

        Deque<String> pathStack = new ArrayDeque<>();
        for (int idx = 0; idx < n; idx++) {
            /**
            (. || empty) nothing
            (..) pop
            (...) push
            (anything) push
            */

            if (p[idx].equals(".") || p[idx].isEmpty()) continue;
            else if (p[idx].equals("..")) {
                if (!pathStack.isEmpty()) {
                    pathStack.pop();
                }
            }
            else {
                pathStack.push(p[idx]);
            }
        }

        if (pathStack.isEmpty()) return "/";

        StringBuilder sb = new StringBuilder();
        Iterator<String> iter = pathStack.descendingIterator();
        while (iter.hasNext()) {
            sb.append("/" + iter.next());
        }

        return sb.toString();
    }
```

```java
class MinStack {
    private int size;
    private int[] stk;
    private int[] minstk;
    private int ptr;

    public MinStack() {
        size = 1024;
        stk = new int[size];
        minstk = new int[size];
        ptr = 0;
    }

    private void expand() {
        int[] t;

        t = new int[size * 2];
        for (int idx = 0; idx < size; idx++) {
            t[idx] = stk[idx];
        }
        stk = t;

        t = new int[size * 2];
        for (int idx = 0; idx< size; idx++) {
            t[idx] = minstk[idx];
        }
        minstk = t;

        size *= 2;
        return;
    }

    public void push(int value) {
        if (ptr == size) {
            expand();
        }

        stk[ptr] = value;
        if (ptr == 0) minstk[ptr] = value;
        else minstk[ptr] = Math.min(value, minstk[ptr - 1]);
        ptr++;

        return;
    }

    public void pop() {
        ptr--;
        return;
    }

    public int top() {
        return stk[ptr - 1];
    }

    public int getMin() {
        return minstk[ptr - 1];
    }
}
```

### [86. Partition List](https://leetcode.com/problems/partition-list/)

```java
/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     int val;
 *     ListNode next;
 *     ListNode() {}
 *     ListNode(int val) { this.val = val; }
 *     ListNode(int val, ListNode next) { this.val = val; this.next = next; }
 * }
 */
class Solution {
    private ListNode detach(ListNode nodeOnLeft) {
        ListNode detachedNode = nodeOnLeft.next;
        ListNode nodeOnRight = detachedNode.next;
        nodeOnLeft.next = nodeOnRight;
        return detachedNode;
    }

    private ListNode attach(ListNode appendingPoint, ListNode target) {
        ListNode nodeRight = appendingPoint.next;
        appendingPoint.next = target;
        target.next = nodeRight;
        return target; // next appending point
    }

    // 'small' nodes should be moved ahead of all the 'big' nodes
    public ListNode partition(ListNode head, int x) {
        // EDGE CASE: null list
        if (head == null) return null;

        // EDGE CASE: no smaller nodes
        ListNode findFirstSmall = head;
        ListNode leftOfFirstSmall = null;
        while (findFirstSmall != null) {
            if (findFirstSmall.val < x) break;
            leftOfFirstSmall = findFirstSmall;
            findFirstSmall = findFirstSmall.next;
        }
        if (findFirstSmall == null) return head;

        // EDGE HANDLING: when list starts with big node, move the first small one to the head
        if (head.val >= x) {
            ListNode detachedNode = detach(leftOfFirstSmall);
            detachedNode.next = head;
            head = detachedNode;
        }

        ListNode appendingPoint = null;
        ListNode bigLeader = head;

        while (bigLeader != null && bigLeader.val < x) {
            appendingPoint = bigLeader;
            bigLeader = bigLeader.next;
        }

        ListNode nodeOnLeft = bigLeader; // nodeOnLeft must be 'big' node
        while (nodeOnLeft != null && nodeOnLeft.next != null) {
            if (nodeOnLeft.next.val >= x) nodeOnLeft = nodeOnLeft.next;
            else {
                ListNode detachedNode = detach(nodeOnLeft);
                appendingPoint = attach(appendingPoint, detachedNode);
            }
        }

        return head;
    }
}
```
