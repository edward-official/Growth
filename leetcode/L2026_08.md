```java
class Solution {
    public int lengthOfLongestSubstring(String s) {
        int n = s.length();
        if (n == 0) return 0;

        HashMap<Character, Boolean> includeChecker = new HashMap<>();
        includeChecker.put(s.charAt(0), true);

        int rvLength = 1;
        int idxL = 0;
        int idxR = 0;
        while (idxL <= idxR) {
            if (idxL >= n || idxR >= n) break;
            if (idxR + 1 == n) break;

            char nextChar = s.charAt(idxR + 1);
            boolean isNextCharIncluded = includeChecker.getOrDefault(nextChar, false);
            if (!isNextCharIncluded) {
                if (++idxR >= n) break;
                includeChecker.put(nextChar, true);

                if (idxR - idxL + 1 > rvLength) {
                    rvLength = idxR - idxL + 1;
                }
            }
            else {
                char removingChar = s.charAt(idxL);
                if (idxL == idxR) {
                    idxL++;
                    includeChecker.put(removingChar, false);
                    idxR++;
                    includeChecker.put(s.charAt(idxR), true);
                }
                else {
                    idxL++;
                    includeChecker.put(removingChar, false);
                }
            }
        }

        return rvLength;
    }
```

```java
class Solution {
    public List<Integer> spiralOrder(int[][] matrix) {
        int borderLeft = 0;
        int borderRight = matrix[0].length - 1;
        int borderTop = 0;
        int borderBottom = matrix.length - 1;

        int[][] step = new int[4][2];
        step[0][0] = 0; step[0][1] = 1;
        step[1][0] = 1; step[1][1] = 0;
        step[2][0] = 0; step[2][1] = -1;
        step[3][0] = -1; step[3][1] = 0;

        List<Integer> rv = new ArrayList<>();
        int direction = 0;
        int r = 0;
        int c = 0;
        rv.add(matrix[r][c]);
        while (borderTop <= borderBottom && borderLeft <= borderRight) {
            while (true) {
                int nextR = r + step[direction][0];
                int nextC = c + step[direction][1];
                if (borderTop <= nextR && nextR <= borderBottom && borderLeft <= nextC && nextC <= borderRight) {
                    r = nextR;
                    c = nextC;
                    rv.add(matrix[r][c]);
                }
                else break;
            }
            if (direction == 0) borderTop++;
            else if (direction == 1) borderRight--;
            else if (direction == 2) borderBottom--;
            else if (direction == 3) borderLeft++;

            direction = (direction + 1) % 4;
        }
        return rv;
    }
}
```

```java
            matrix[base + sz - 1 - offset][base] = t3;
            matrix[base][base + offset] = t4;
        }
    }

        .
        .
        .
        (n - 1) ... (n - 1)
         */

            processFourLines(matrix, sz);
        }

    }
```

### [289. Game of Life](https://leetcode.com/problems/game-of-life/)

```java
class Solution {
    private int[][] updatingBoard;

    private int getNumberOfLiveFriends(int[][] board, int r, int c) {
        int m = board.length;
        int n = board[0].length;

        int rv = 0;
        if (0 <= r - 1 && r - 1 < m && board[r - 1][c] == 1) rv++;
        if (0 <= r + 1 && r + 1 < m && board[r + 1][c] == 1) rv++;
        if (0 <= c - 1 && c - 1 < n && board[r][c - 1] == 1) rv++;
        if (0 <= c + 1 && c + 1 < n && board[r][c + 1] == 1) rv++;
        if (0 <= r - 1 && r - 1 < m && 0 <= c - 1 && c - 1 < n && board[r - 1][c - 1] == 1) rv++;
        if (0 <= r - 1 && r - 1 < m && 0 <= c + 1 && c + 1 < n && board[r - 1][c + 1] == 1) rv++;
        if (0 <= r + 1 && r + 1 < m && 0 <= c - 1 && c - 1 < n && board[r + 1][c - 1] == 1) rv++;
        if (0 <= r + 1 && r + 1 < m && 0 <= c + 1 && c + 1 < n && board[r + 1][c + 1] == 1) rv++;

        return rv;
    }

    private void proceedOneGeneration(int[][] board) {
        int m = board.length;
        int n = board[0].length;

        for (int r = 0; r < m; r++) {
            for (int c = 0; c < n; c++) {
                int cnt = getNumberOfLiveFriends(board, r, c);
                if (cnt == 2) updatingBoard[r][c] = board[r][c];
                else if (cnt == 3) updatingBoard[r][c] = 1;
                else updatingBoard[r][c] = 0;
            }
        }

        for (int r = 0; r < m; r++) {
            for (int c = 0; c < n; c++) {
                board[r][c] = updatingBoard[r][c];
            }
        }

        return;
    }

    public void gameOfLife(int[][] board) {
        // 2 live neighbors: live
        // 3 live neighbors: live + reproduction
        int m = board.length;
        int n = board[0].length;

        updatingBoard = new int[m][n];
        proceedOneGeneration(board);

        return;
    }
}
```

### [290. Word Pattern](https://leetcode.com/problems/word-pattern/)

```java
class Solution {
    private int findOpeningIdx(String target, int idx) {
        while (target.charAt(idx) == ' ') {
            idx++;
            if (target.length() <= idx) {
                return -1;
            }
        }
        return idx;
    }

    private int findClosingIdx(String target, int idx) {
        while (target.charAt(idx) != ' ') {
            idx++;
            if (target.length() == idx) {
                return idx;
            }
        }
        return idx;
    }

    private String getWord(String target, int idx) {
        int openingIdx = findOpeningIdx(target, idx);
        int closingIdx = findClosingIdx(target, openingIdx);

        return target.substring(openingIdx, closingIdx);
    }

    public boolean wordPattern(String pattern, String target) {
        Map<Character, String> ptMap = new HashMap<>();
        Map<String, Character> tpMap = new HashMap<>();

        int idxPattern = 0;
        int idxTarget = 0;
        while (idxPattern < pattern.length() && idxTarget < target.length()) {
            int nextIdxTarget = findClosingIdx(target, findOpeningIdx(target, idxTarget));
            char p = pattern.charAt(idxPattern++);
            String w = getWord(target, idxTarget);

            if (ptMap.containsKey(p) && !ptMap.get(p).equals(w)) {
                return false;
            }
            else if (!ptMap.containsKey(p)) {
                ptMap.put(p, w);
            }

            if (tpMap.containsKey(w) && !tpMap.get(w).equals(p)) {
                return false;
            }
            else if (!tpMap.containsKey(w)) {
                tpMap.put(w, p);
            }

            idxTarget = nextIdxTarget;
        }

        if (idxPattern < pattern.length() || idxTarget < target.length()) {
            return false;
        }
        return true;
    }
}
```

```java
            else {
                addRange(rv, nums, openingIdx, closingIdx);
                openingIdx = idx;
                closingIdx = idx;
            }
        }
        addRange(rv, nums, openingIdx, closingIdx);

        return rv;
    }
}
```

### [56. Merge Intervals](https://leetcode.com/problems/merge-intervals/)

```java
class Solution {
    private void sortIntervals(int[][] intervals) {
        Arrays.sort(intervals, (a, b) -> Integer.compare(a[0], b[0]));

        return;
    }

    public int[][] merge(int[][] intervals) {
        if (intervals.length <= 1) return intervals;

        sortIntervals(intervals);

        List<int[]> merged = new ArrayList<>();
        int[] currentInterval = intervals[0];
        merged.add(currentInterval);

        for (int[] interval: intervals) {
            int currentEnd = currentInterval[1];
            int nextStart = interval[0];
            int nextEnd = interval[1];

            if (nextStart <= currentEnd) {
                currentInterval[1] = Math.max(currentEnd, nextEnd);
            }
            else {
                currentInterval = interval;
                merged.add(currentInterval);
            }
        }

        return merged.toArray(new int[merged.size()][]);
    }
}
```

### [57. Insert Interval](https://leetcode.com/problems/insert-interval/)

```java
class Solution {
    public int[][] insert(int[][] intervals, int[] newInterval) {
        int minRange = newInterval[0];
        int maxRange = newInterval[1];

        List<int[]> rv = new ArrayList<>();
        int idx = 0;
        int n = intervals.length;

        while (idx < n && intervals[idx][1] < minRange) {
            rv.add(intervals[idx]);
            idx++;
        }

        while (idx < n && intervals[idx][0] <= maxRange) {
            newInterval[0] = Math.min(newInterval[0], intervals[idx][0]);
            newInterval[1] = Math.max(newInterval[1], intervals[idx][1]);
            idx++;
        }
        rv.add(newInterval);

        while (idx < n) {
            rv.add(intervals[idx]);
            idx++;
        }

        return rv.toArray(new int[rv.size()][]);
    }
}
```
