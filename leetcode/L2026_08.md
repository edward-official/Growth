```java
class Solution {
    public boolean isSubsequence(String s, String t) {
        int ns = s.length();
        int nt = t.length();
        int ps = 0;
        int pt = 0;

        while (ps < ns) {
            while (pt < nt && t.charAt(pt) != s.charAt(ps)) {
                pt++;
            }
            if (pt == nt) return false;
            ps++;
            pt++;
        }

        if (ps == ns) return true;
        return false;
    }
}
```

```java
class Solution {
    private void findTarget(int[] nums, int openingIdx, int target, List<List<Integer>> rv) {
        int idxL = openingIdx;
        int idxR = nums.length - 1;

        while (idxL < idxR) {
            int t = nums[idxL] + nums[idxR];
            if (t == target) {
                rv.add(Arrays.asList(-target, nums[idxL], nums[idxR]));
                while (idxL < idxR && nums[idxL] == nums[idxL + 1]) idxL++;
                while (idxL < idxR && nums[idxR] == nums[idxR - 1]) idxR--;

                idxL++;
                idxR--;
            }
            else if (t < target) idxL++;
            else idxR--;
        }
    }

    public List<List<Integer>> threeSum(int[] nums) {
        Arrays.sort(nums);
        int n = nums.length;

        List<List<Integer>> rv = new ArrayList<>();
        for (int idx = 0; idx < n; idx++) {
            int item = nums[idx];
            if (item > 0) break;
            if (idx > 0 && nums[idx - 1] == nums[idx]) continue;
            findTarget(nums, idx + 1, -item, rv);
        }

        return rv;
    }
}
```

### [209. Minimum Size Subarray Sum](https://leetcode.com/problems/minimum-size-subarray-sum/)

```java
class Solution {
    public int minSubArrayLen(int target, int[] nums) {
        int n = nums.length;

        int idxL = 0;
        int idxR = 0;
        int currentSum = nums[0];
        int minLength = n + 1;
        while (idxL < n && idxR < n) {
            if (idxL == idxR) {
                if (currentSum >= target) return 1;
                else if (++idxR < n) currentSum += nums[idxR];
            }
            else {
                if (currentSum < target && ++idxR < n) currentSum += nums[idxR];
                else {
                    if (idxR - idxL + 1 < minLength) {
                        minLength = idxR - idxL + 1;
                    }
                    currentSum -= nums[idxL++];
                }
            }
        }

        if (minLength == n + 1) return 0;
        return minLength;
    }
}
```

### [3. Longest Substring Without Repeating Characters](https://leetcode.com/problems/longest-substring-without-repeating-characters/)

```java
class Solution {
    public int lengthOfLongestSubstring(String s) {
        int n = s.length();
        if (n == 0) return 0;

        HashMap<Character, Boolean> includeChecker = new HashMap<>();
        includeChecker.put(s.charAt(0), true);

        int rvLength = 1;
        int idxL = 0;
        int idxR = 0;
        while (idxL <= idxR) {
            if (idxL >= n || idxR >= n) break;
            if (idxR + 1 == n) break;

            char nextChar = s.charAt(idxR + 1);
            boolean isNextCharIncluded = includeChecker.getOrDefault(nextChar, false);
            if (!isNextCharIncluded) {
                if (++idxR >= n) break;
                includeChecker.put(nextChar, true);

                if (idxR - idxL + 1 > rvLength) {
                    rvLength = idxR - idxL + 1;
                }
            }
            else {
                char removingChar = s.charAt(idxL);
                if (idxL == idxR) {
                    idxL++;
                    includeChecker.put(removingChar, false);
                    idxR++;
                    includeChecker.put(s.charAt(idxR), true);
                }
                else {
                    idxL++;
                    includeChecker.put(removingChar, false);
                }
            }
        }

        return rvLength;
    }
}
```

### [54. Spiral Matrix](https://leetcode.com/problems/spiral-matrix/)

```java
class Solution {
    public List<Integer> spiralOrder(int[][] matrix) {
        int borderLeft = 0;
        int borderRight = matrix[0].length - 1;
        int borderTop = 0;
        int borderBottom = matrix.length - 1;

        int[][] step = new int[4][2];
        step[0][0] = 0; step[0][1] = 1;
        step[1][0] = 1; step[1][1] = 0;
        step[2][0] = 0; step[2][1] = -1;
        step[3][0] = -1; step[3][1] = 0;

        List<Integer> rv = new ArrayList<>();
        int direction = 0;
        int r = 0;
        int c = 0;
        rv.add(matrix[r][c]);
        while (borderTop <= borderBottom && borderLeft <= borderRight) {
            while (true) {
                int nextR = r + step[direction][0];
                int nextC = c + step[direction][1];
                if (borderTop <= nextR && nextR <= borderBottom && borderLeft <= nextC && nextC <= borderRight) {
                    r = nextR;
                    c = nextC;
                    rv.add(matrix[r][c]);
                }
                else break;
            }
            if (direction == 0) borderTop++;
            else if (direction == 1) borderRight--;
            else if (direction == 2) borderBottom--;
            else if (direction == 3) borderLeft++;

            direction = (direction + 1) % 4;
        }
        return rv;
    }
}
```

```java
class Solution {
    private void processFourLines(int[][] matrix, int sz) {
        if (sz == 1 || sz == 0) return;
        /**
        matrix: 11 x 11
        size: 11 -> 9 -> "7"
        starting point: (2,2)
        number of repetition: 6
         */

        int base = (matrix.length - sz) / 2;
        for (int offset = 0; offset < sz - 1; offset++) {
            int t1 = matrix[base][base + offset];
            int t2 = matrix[base + offset][base + sz - 1];
            int t3 = matrix[base + sz - 1][base + sz - 1 - offset];
            int t4 = matrix[base + sz - 1 - offset][base];

            matrix[base + offset][base + sz - 1] = t1;
            matrix[base + sz - 1][base + sz - 1 - offset] = t2;
            matrix[base + sz - 1 - offset][base] = t3;
            matrix[base][base + offset] = t4;
        }
    }

    public void rotate(int[][] matrix) {
        /**
        0 ... (n - 1)
        .
        .
        .
        (n - 1) ... (n - 1)
         */

        for (int sz = matrix.length; sz > 0; sz -= 2) {
            processFourLines(matrix, sz);
        }

        return;
    }
}
```

```java
class Solution {
    private boolean flagFirstRow = false;
    private boolean flagFirstColumn = false;

    private void processSpecialFlags(int[][] matrix) {
        int m = matrix.length;
        int n = matrix[0].length;

        if (matrix[0][0] == 0) {
            flagFirstRow = true;
            flagFirstColumn = true;
            return;
        }

        for (int r = 1; r < m; r++) {
            if (matrix[r][0] == 0) {
                flagFirstColumn = true;
                break;
            }
        }

        for (int c = 1; c < n; c++) {
            if (matrix[0][c] == 0) {
                flagFirstRow = true;
                break;
            }
        }
        
        return;
    }

    private void processZeros(int[][] matrix) {
        int m = matrix.length;
        int n = matrix[0].length;

        for (int r = 0; r < m; r++) {
            for (int c = 0; c < n; c++) {
                if (matrix[r][c] == 0) {
                    matrix[r][0] = 0;
                    matrix[0][c] = 0;
                }
            }
        }

        return;
    }

    private void processFlags(int[][] matrix) {
        int m = matrix.length;
        int n = matrix[0].length;

        for (int r = 1; r < m; r++) {
            if (matrix[r][0] != 0) continue;
            for (int c = 1; c < n; c++) {
                matrix[r][c] = 0;
            }
        }

        for (int c = 1; c < n; c++) {
            if (matrix[0][c] != 0) continue;
            for (int r = 1; r < m; r++) {
                matrix[r][c] = 0;
            }
        }

        if (flagFirstRow) {
            for (int c = 0; c < n; c++) {
                matrix[0][c] = 0;
            }
        }

        if (flagFirstColumn) {
            for (int r = 0; r < m; r++) {
                matrix[r][0] = 0;
            }
        }

        return;
    }

    public void setZeroes(int[][] matrix) {
        processSpecialFlags(matrix);
        processZeros(matrix);
        processFlags(matrix);
        return;
    }
}
```

### [289. Game of Life](https://leetcode.com/problems/game-of-life/)

```java
class Solution {
    private int[][] updatingBoard;

    private int getNumberOfLiveFriends(int[][] board, int r, int c) {
        int m = board.length;
        int n = board[0].length;

        int rv = 0;
        if (0 <= r - 1 && r - 1 < m && board[r - 1][c] == 1) rv++;
        if (0 <= r + 1 && r + 1 < m && board[r + 1][c] == 1) rv++;
        if (0 <= c - 1 && c - 1 < n && board[r][c - 1] == 1) rv++;
        if (0 <= c + 1 && c + 1 < n && board[r][c + 1] == 1) rv++;
        if (0 <= r - 1 && r - 1 < m && 0 <= c - 1 && c - 1 < n && board[r - 1][c - 1] == 1) rv++;
        if (0 <= r - 1 && r - 1 < m && 0 <= c + 1 && c + 1 < n && board[r - 1][c + 1] == 1) rv++;
        if (0 <= r + 1 && r + 1 < m && 0 <= c - 1 && c - 1 < n && board[r + 1][c - 1] == 1) rv++;
        if (0 <= r + 1 && r + 1 < m && 0 <= c + 1 && c + 1 < n && board[r + 1][c + 1] == 1) rv++;

        return rv;
    }

    private void proceedOneGeneration(int[][] board) {
        int m = board.length;
        int n = board[0].length;

        for (int r = 0; r < m; r++) {
            for (int c = 0; c < n; c++) {
                int cnt = getNumberOfLiveFriends(board, r, c);
                if (cnt == 2) updatingBoard[r][c] = board[r][c];
                else if (cnt == 3) updatingBoard[r][c] = 1;
                else updatingBoard[r][c] = 0;
            }
        }

        for (int r = 0; r < m; r++) {
            for (int c = 0; c < n; c++) {
                board[r][c] = updatingBoard[r][c];
            }
        }

        return;
    }

    public void gameOfLife(int[][] board) {
        // 2 live neighbors: live
        // 3 live neighbors: live + reproduction
        int m = board.length;
        int n = board[0].length;

        updatingBoard = new int[m][n];
        proceedOneGeneration(board);

        return;
    }
}
```

### [383. Ransom Note](https://leetcode.com/problems/ransom-note/)

```java
class Solution {
    public boolean canConstruct(String ransomNote, String magazine) {
        if (ransomNote.length() > magazine.length()) {
            return false;
        }

        Map<Character, Integer> countMap = new HashMap<>();

        int nMagazine = magazine.length();
        for (int idx = 0; idx < nMagazine; idx++) {
            char k = magazine.charAt(idx);
            int v = countMap.getOrDefault(k, 0);
            countMap.put(k, v + 1);
        }

        int nRansomNote = ransomNote.length();
        for (int idx = 0; idx < nRansomNote; idx++) {
            char k = ransomNote.charAt(idx);
            int v = countMap.getOrDefault(k, 0);
            if (v == 0) {
                return false;
            }
            countMap.put(k, v - 1);
        }

        return true;
    }
}
```
