### [125. Valid Palindrome](https://leetcode.com/problems/valid-palindrome/)

```java
class Solution {
    public boolean isPalindrome(String s) {
        int n = s.length();

        StringBuilder sb = new StringBuilder();
        for (int idx = 0; idx < n; idx++) {
            char tmp = s.charAt(idx);
            if (Character.isLetterOrDigit(tmp)) {
                sb.append(Character.toLowerCase(tmp));
            }
        }

        String t1 = sb.toString();
        String t2 = sb.reverse().toString();

        return t1.equals(t2);
    }
}
```

```java
class Solution {
    public boolean isPalindrome(String s) {
        int pLeft = 0;
        int pRight = s.length() - 1;

        while (pLeft < pRight) {
            while (pLeft < s.length() && !Character.isLetterOrDigit(s.charAt(pLeft))) {
                pLeft++;
            }
            while (0 <= pRight && !Character.isLetterOrDigit(s.charAt(pRight))) {
                pRight--;
            }
            if (pLeft >= pRight) return true;
            if (Character.toLowerCase(s.charAt(pLeft)) == Character.toLowerCase(s.charAt(pRight))) {
                pLeft++;
                pRight--;
            }
            else return false;
        }

        return true;
    }
}
```

### [392. Is Subsequence](https://leetcode.com/problems/is-subsequence/)

```java
class Solution {
    public boolean isSubsequence(String s, String t) {
        int ns = s.length();
        int nt = t.length();
        int ps = 0;
        int pt = 0;

        while (ps < ns) {
            while (pt < nt && t.charAt(pt) != s.charAt(ps)) {
                pt++;
            }
            if (pt == nt) return false;
            ps++;
            pt++;
        }

        if (ps == ns) return true;
        return false;
    }
}
```

### [167. Two Sum II - Input Array Is Sorted](https://leetcode.com/problems/two-sum-ii-input-array-is-sorted/)

```java
class Solution {
    public int[] twoSum(int[] numbers, int target) {
        int idxLeft = 0;
        int idxRight = numbers.length - 1;

        while (idxLeft < idxRight) {
            int t = numbers[idxLeft] + numbers[idxRight];
            if (t == target) break;
            if (t < target) idxLeft++;
            if (t > target) idxRight--;
        }

        int[] rv = new int[2];
        rv[0] = idxLeft + 1;
        rv[1] = idxRight + 1;
        return rv;
    }
}
```

### [11. Container With Most Water](https://leetcode.com/problems/container-with-most-water/)

```java
class Solution {
    public int maxArea(int[] height) {
        int idxL = 0;
        int idxR = height.length - 1;

        int rv = 0;
        while (idxL < idxR) {
            int t = Math.min(height[idxL], height[idxR]) * (idxR - idxL);
            if (rv < t) rv = t;
            if (height[idxL] < height[idxR]) idxL++;
            else idxR--;
        }

        return rv;
    }
}
```

### [15. 3Sum](https://leetcode.com/problems/3sum/)

```java
class Solution {
    private void findTarget(int[] nums, int openingIdx, int target, List<List<Integer>> rv) {
        int idxL = openingIdx;
        int idxR = nums.length - 1;

        while (idxL < idxR) {
            int t = nums[idxL] + nums[idxR];
            if (t == target) {
                rv.add(Arrays.asList(-target, nums[idxL], nums[idxR]));
                while (idxL < idxR && nums[idxL] == nums[idxL + 1]) idxL++;
                while (idxL < idxR && nums[idxR] == nums[idxR - 1]) idxR--;

                idxL++;
                idxR--;
            }
            else if (t < target) idxL++;
            else idxR--;
        }
    }

    public List<List<Integer>> threeSum(int[] nums) {
        Arrays.sort(nums);
        int n = nums.length;
        
        List<List<Integer>> rv = new ArrayList<>();
        for (int idx = 0; idx < n; idx++) {
            int item = nums[idx];
            if (item > 0) break;
            if (idx > 0 && nums[idx - 1] == nums[idx]) continue;
            findTarget(nums, idx + 1, -item, rv);
        }

        return rv;
    }
}
```

### [209. Minimum Size Subarray Sum](https://leetcode.com/problems/minimum-size-subarray-sum/)

```java
class Solution {
    public int minSubArrayLen(int target, int[] nums) {
        int n = nums.length;

        int idxL = 0;
        int idxR = 0;
        int currentSum = nums[0];
        int minLength = n + 1;
        while (idxL < n && idxR < n) {
            if (idxL == idxR) {
                if (currentSum >= target) return 1;
                else if (++idxR < n) currentSum += nums[idxR];
            }
            else {
                if (currentSum < target && ++idxR < n) currentSum += nums[idxR];
                else {
                    if (idxR - idxL + 1 < minLength) {
                        minLength = idxR - idxL + 1;
                    }
                    currentSum -= nums[idxL++];
                }
            }
        }

        if (minLength == n + 1) return 0;
        return minLength;
    }
}
```

### [3. Longest Substring Without Repeating Characters](https://leetcode.com/problems/longest-substring-without-repeating-characters/)

```java
class Solution {
    public int lengthOfLongestSubstring(String s) {
        int n = s.length();
        if (n == 0) return 0;

        HashMap<Character, Boolean> includeChecker = new HashMap<>();
        includeChecker.put(s.charAt(0), true);

        int rvLength = 1;
        int idxL = 0;
        int idxR = 0;
        while (idxL <= idxR) {
            if (idxL >= n || idxR >= n) break;
            if (idxR + 1 == n) break;

            char nextChar = s.charAt(idxR + 1);
            boolean isNextCharIncluded = includeChecker.getOrDefault(nextChar, false);
            if (!isNextCharIncluded) {
                if (++idxR >= n) break;
                includeChecker.put(nextChar, true);

                if (idxR - idxL + 1 > rvLength) {
                    rvLength = idxR - idxL + 1;
                }
            }
            else {
                char removingChar = s.charAt(idxL);
                if (idxL == idxR) {
                    idxL++;
                    includeChecker.put(removingChar, false);
                    idxR++;
                    includeChecker.put(s.charAt(idxR), true);
                }
                else {
                    idxL++;
                    includeChecker.put(removingChar, false);
                }
            }
        }

        return rvLength;
    }
}
```
