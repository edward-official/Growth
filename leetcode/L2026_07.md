### [1. Two Sum](https://leetcode.com/problems/two-sum/)

```java
class Solution {
    public int[] twoSum(int[] nums, int target) {
        Map<Integer, Integer> finder = new HashMap<>(); // value of the number -> index

        int n = nums.length;
        int complement;
        int[] rv = new int[2];
        for (int idx = 0; idx < n; idx++) {
            complement = target - nums[idx];

            if (finder.containsKey(complement)) {
                rv[0] = finder.get(complement);
                rv[1] = idx;
                return rv;
            }

            finder.put(nums[idx], idx);
        }

        return rv;
    }
}
```

```java
class Solution {
    public int maxArea(int[] height) {
        int idxL = 0;
        int idxR = height.length - 1;

        int rv = 0;
        int tmp;
        while (idxL < idxR) {
            tmp = (idxR - idxL) * Math.min(height[idxL], height[idxR]);
            rv = Math.max(rv, tmp);

            if (height[idxL] < height[idxR]) idxL++;
            else idxR--;
        }

        return rv;
    }
}
```

### [2224. Minimum Number of Operations to Convert Time](https://leetcode.com/problems/minimum-number-of-operations-to-convert-time/)

```java
class Solution {
    public int convertTime(String current, String correct) {
        int currentH = Integer.parseInt(current.substring(0, 2));
        int currentM = Integer.parseInt(current.substring(3, 5));
        int currentT = currentH * 60 + currentM;

        int correctH = Integer.parseInt(correct.substring(0, 2));
        int correctM = Integer.parseInt(correct.substring(3, 5));
        int correctT = correctH * 60 + correctM;

        int tmp = Math.max(currentT - correctT, correctT - currentT);
        int rv = 0;
        int[] options = {60, 15, 5, 1};
        for (int option: options) {
            rv += tmp / option;
            tmp %= option;
        }

        return rv;
    }
}
```

```java
class Solution {
    public String removeDigit(String number, char digit) {
        int n = number.length();
        int idxFinder = -1;

        for (int idx = 0; idx < n; idx++) {
            if (number.charAt(idx) == digit) {
                if (idx + 1 < n && number.charAt(idx + 1) > digit) {
                    idxFinder = idx;
                    break;
                }
                idxFinder = idx;
            }
        }

        return number.substring(0, idxFinder) + number.substring(idxFinder + 1, n);
    }
}
```

### [134. Gas Station](https://leetcode.com/problems/gas-station/)

```java
class Solution {
    public int canCompleteCircuit(int[] gas, int[] cost) {
        int n = gas.length;
        int[] net = new int[n];
        for (int idx = 0; idx < n; idx++) {
            net[idx] = gas[idx] - cost[idx];
        }

        // IMPOSSIBLE: the amount of gas isn't enought to pay all the costs of the stations
        int totalNet = Arrays.stream(net).sum();
        if (totalNet < 0) return -1;

        // POSSIBLE: the amount of gas is enough to pay all the costs of the stations
        int balance = 0;
        int startingIdx = 0;
        int idx = 0;

        while (idx < n) {
            balance += net[idx];
            if (balance < 0) {
                startingIdx = idx + 1;
                balance = 0;
            }
            idx++;
        }

        return startingIdx;
    }
}
```

### [179. Largest Number](https://leetcode.com/problems/largest-number/)

```java
class Solution {
    public String largestNumber(int[] nums) {
        int n = nums.length;
        String[] sNums = new String[n];
        for (int idx = 0; idx < n; idx++) {
            sNums[idx] = String.valueOf(nums[idx]);
        }

        Arrays.sort(sNums, (a, b) -> (b + a).compareTo(a + b));
        if (sNums[0].equals("0")) return "0";

        StringBuilder sb = new StringBuilder();
        for (String sNum: sNums) {
            sb.append(sNum);
        }

        return sb.toString();
    }
}
```

### [316. Remove Duplicate Letters](https://leetcode.com/problems/remove-duplicate-letters/)

```java
class Solution {
    public String removeDuplicateLetters(String s) {
        int n = s.length();

        // First, count the number of each alphabet
        HashMap<Character, Integer> countMap = new HashMap<>();
        for (int idx = 0; idx < n; idx++) {
            char tmp = s.charAt(idx);
            countMap.put(tmp, countMap.getOrDefault(tmp, 0) + 1);
        }

        // Then, traverse and remove the letters
        HashMap<Character, Boolean> checkUsed = new HashMap<>();
        Stack<Character> rv = new Stack<>();
        for (int idx = 0; idx < n; idx++) {
            char tmp = s.charAt(idx);
            countMap.put(tmp, countMap.get(tmp) - 1);

            // CASE 1: continue
            if (checkUsed.getOrDefault(tmp, false)) continue;

            // CASE 2: append
            while (!rv.isEmpty() && rv.peek() > tmp && countMap.get(rv.peek()) > 0) {
                char removed = rv.pop();
                checkUsed.put(removed, false);
            }
            rv.push(tmp);
            checkUsed.put(tmp, true);
        }

        StringBuilder sb = new StringBuilder();
        for (char c: rv) sb.append(c);
        return sb.toString();
    }
}
```

### [324. Wiggle Sort II](https://leetcode.com/problems/wiggle-sort-ii/)

```java
class Solution {
    public void wiggleSort(int[] nums) {
        int n = nums.length;

        int[] sortedNums = nums.clone();
        Arrays.sort(sortedNums);

        int idxL = (n - 1) / 2;
        int idxR = n - 1;
        for (int idx = 0; idx < n; idx++) {
            if (idx % 2 == 0) nums[idx] = sortedNums[idxL--];
            else nums[idx] = sortedNums[idxR--];
        }
    }
}
```

### [88. Merge Sorted Array](https://leetcode.com/problems/merge-sorted-array/)

```java
class Solution {
    public void merge(int[] nums1, int m, int[] nums2, int n) {
        Stack<Integer> merged = new Stack<>();
        int idx1 = 0;
        int idx2 = 0;

        while (idx1 < m && idx2 < n) {
            if (nums1[idx1] < nums2[idx2]) {
                merged.push(nums1[idx1]);
                idx1++;
            }
            else {
                merged.push(nums2[idx2]);
                idx2++;
            }
        }

        while (idx1 < m) {
            merged.push(nums1[idx1]);
            idx1++;
        }
        while (idx2 < n) {
            merged.push(nums2[idx2]);
            idx2++;
        }

        for (int idx = m + n - 1; idx >= 0; idx--) {
            nums1[idx] = merged.get(idx);
        }
    }
}
```

### [27. Remove Element](https://leetcode.com/problems/remove-element/)

```java
class Solution {
    public int removeElement(int[] nums, int target) {
        int n = nums.length;
        int nRemoved = 0;
        int idx = 0;

        while (idx < n - nRemoved) {
            if (nums[idx] == target) {
                nRemoved++;
                nums[idx] = nums[n - nRemoved];
                nums[n - nRemoved] = -1;
            }
            else idx++;
        }

        return (n - nRemoved);
    }
}
```

### [26. Remove Duplicates from Sorted Array](https://leetcode.com/problems/remove-duplicates-from-sorted-array/)

```java
class Solution {
    public int removeDuplicates(int[] nums) {
        int n = nums.length;
        int rv = 0;
        Stack<Integer> uniqueNums = new Stack<>();

        for (int idx = 0; idx < n; idx++) {
            if (uniqueNums.isEmpty() || uniqueNums.peek() != nums[idx]) {
                uniqueNums.push(nums[idx]);
                rv++;
            }
        }

        int nUniqueNums = uniqueNums.size();
        for (int idx = 0; idx < nUniqueNums; idx++) {
            nums[idx] = uniqueNums.get(idx);
        }

        return rv;
    }
}
```

### [80. Remove Duplicates from Sorted Array II](https://leetcode.com/problems/remove-duplicates-from-sorted-array-ii/)

```java
class Solution {
    public int removeDuplicates(int[] nums) {
        int n = nums.length;
        int currentItem = nums[0] - 1;
        int currentCounter = 0;
        int updatingIdx = 0;

        for (int idx = 0; idx < n; idx++) {
            if (currentItem != nums[idx]) {
                currentItem = nums[idx];
                currentCounter = 1;
                nums[updatingIdx++] = currentItem;
            }
            else if (currentCounter == 1) {
                currentCounter++;
                nums[updatingIdx++] = currentItem;
            }
        }

        return updatingIdx;
    }
}
```

### [169. Majority Element](https://leetcode.com/problems/majority-element/)

```java
class Solution {
    public int majorityElement(int[] nums) {
        int n = nums.length;

        HashMap<Integer, Integer> cnt = new HashMap<>();
        for (int idx = 0; idx < n; idx++) {
            int k = nums[idx];
            int v = 1;

            if (cnt.containsKey(k)) {
                v += cnt.get(k);
            }

            if (v >= (n + 1) / 2) {
                return k;
            }
            cnt.put(k, v);
        }

        return -1;
    }
}
```

### [189. Rotate Array](https://leetcode.com/problems/rotate-array/)

```java
class Solution {
    public void rotate(int[] nums, int k) {
        int n = nums.length;

        k %= n;
        if (k == 0) {
            return;
        }

        int[] tmp = new int[n];
        for (int idx = 0; idx < n - k; idx++) {
            tmp[idx + k] = nums[idx];
        }
        for (int idx = n - k; idx < n; idx++) {
            tmp[idx + k - n] = nums[idx];
        }

        for (int idx = 0; idx < n; idx++) {
            nums[idx] = tmp[idx];
        }
        return;
    }
}
```

```java
class Solution {
    public boolean canJump(int[] nums) {
        int n = nums.length;
        int maxReach = 0 + nums[0];

        for (int idx = 0; idx <= maxReach && idx < n; idx++) {
            if (idx + nums[idx] > maxReach) {
                maxReach = idx + nums[idx];
            }
        }

        if (maxReach >= n - 1) return true;
        return false;
    }
}
```

### [45. Jump Game II](https://leetcode.com/problems/jump-game-ii/)

```java
class Solution {
    public int jump(int[] nums) {
        int n = nums.length;
        if (n == 1) return 0;

        int cnt = 0;
        int tempMax = 0;
        int nextMax = -1;

        for (int idx = 0; idx < n; idx++) {
            // update 'nextMax'
            if (idx <= tempMax) {
                nextMax = Math.max(nextMax, idx + nums[idx]);
                if (nextMax >= n - 1) {
                    return cnt + 1;
                }
            }

            if (idx == tempMax && tempMax < nextMax) {
                if (tempMax + 1 < n) {
                    cnt++;
                    tempMax = nextMax;
                }
                if (tempMax + 1 >= n) {
                    break;
                }
            }
        }

        return -1;
    }
}
```

### [274. H-Index](https://leetcode.com/problems/h-index/)

```java
class Solution {
    public int hIndex(int[] citations) {
        int n = citations.length;
        HashMap<Integer, Integer> getNumberOfPapers = new HashMap<>();
        int nCiteMax = 0;

        for (int idx = 0; idx < n; idx++) {
            if (nCiteMax < citations[idx]) {
                nCiteMax = citations[idx];
            }

            getNumberOfPapers.put(citations[idx], getNumberOfPapers.getOrDefault(citations[idx], 0) + 1);
        }

        int cnt = 0;
        for (int h = nCiteMax; h >= 0; h--) {
            cnt += getNumberOfPapers.getOrDefault(h, 0);
            if (cnt >= h) {
                return h;
            }
        }

        return -1;
    }
}
```

### [380. Insert Delete GetRandom O(1)](https://leetcode.com/problems/insert-delete-getrandom-o1/)

```java
class RandomizedSet {
    private List<Integer> items;
    private Map<Integer, Integer> getIdx;
    private Random random;

    public RandomizedSet() {
        items = new ArrayList<>();
        getIdx = new HashMap<>();
        random = new Random();
    }
    
    public boolean insert(int val) {
        if (getIdx.getOrDefault(val, -1) == -1) {
            items.add(val);
            getIdx.put(val, items.size() - 1);
            return true;
        }
        return false;
    }
    
    public boolean remove(int val) {
        int valIdx = getIdx.getOrDefault(val, -1);

        if (valIdx != -1) {
            if (valIdx != items.size() - 1) {
                int movingItem = items.get(items.size() - 1);
                items.set(valIdx, movingItem);
                getIdx.put(movingItem, valIdx);
            }
            items.remove(items.size() - 1);
            getIdx.remove(val);
            return true;
        }
        return false;
    }
    
    public int getRandom() {
        int rIdx = random.nextInt(items.size());
        return items.get(rIdx);
    }
}
```
