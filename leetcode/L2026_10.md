```java
class Solution {
    public int maxSubarraySumCircular(int[] nums) {
        int regionMax = nums[0];
        int regionMin = nums[0];

        int globalMax = regionMax;
        int globalMin = regionMin;

        for (int idx = 1; idx < nums.length; idx++) {
            regionMax = Math.max(nums[idx], regionMax + nums[idx]);
            regionMin = Math.min(nums[idx], regionMin + nums[idx]);

            if (globalMax < regionMax) globalMax = regionMax;
            if (globalMin > regionMin) globalMin = regionMin;
        }

        if (globalMax < 0) return globalMax;
        return Math.max(globalMax, Arrays.stream(nums).sum() - globalMin);
    }
}
```

```java
class Solution {
    private int findRow(int[][] matrix, int target) {
        int rangeL = 0;
        int rangeR = matrix.length - 1;

        while (rangeL <= rangeR) {
            int middleIdx = (rangeL + rangeR) / 2;

            int minItem = matrix[middleIdx][0];
            int maxItem = matrix[middleIdx][matrix[0].length - 1];

            if (minItem <= target && target <= maxItem) return middleIdx;
            else if (target < minItem) rangeR = middleIdx - 1;
            else rangeL = middleIdx + 1;
        }

        return -1;
    }

    private int findColumn(int[] row, int target) {
        int rangeL = 0;
        int rangeR = row.length - 1;

        while (rangeL <= rangeR) {
            int middleIdx = (rangeL + rangeR) / 2;
            int middleItem = row[middleIdx];
            if (middleItem == target) return middleIdx;
            else if (middleItem < target) rangeL = middleIdx + 1;
            else rangeR = middleIdx - 1;
        }

        return -1;
    }

    public boolean searchMatrix(int[][] matrix, int target) {
        int row = findRow(matrix, target);
        if (row == -1) return false;

        int column = findColumn(matrix[row], target);
        return (column != -1);
    }
}
```

```java
class Solution {
    private boolean isPeak(int[] nums, int idx) {
        if (idx == 0) {
            if (nums[idx] > nums[idx + 1]) return true;
            return false;
        }

        if (idx == nums.length - 1) {
            if (nums[idx] > nums[idx - 1]) return true;
            return false;
        }

        if (nums[idx - 1] < nums[idx] && nums[idx + 1] < nums[idx]) return true;
        return false;
    }

    public int findPeakElement(int[] nums) {
        if (nums.length == 1) return 0;
        if (nums.length == 2) {
            if (nums[0] > nums[1]) return 0;
            else return 1;
        }

        int rangeL = 0;
        int rangeR = nums.length - 1;

        while (rangeL <= rangeR) {
            int mid = (rangeL + rangeR) / 2;
            if (isPeak(nums, mid)) return mid;

            if (nums[mid] < nums[mid + 1]) rangeL = mid + 1;
            else rangeR = mid - 1;
        }

        return -1;
    }
}
```

### [33. Search in Rotated Sorted Array](https://leetcode.com/problems/search-in-rotated-sorted-array/)

```java
class Solution {
    private boolean isPartSorted(int[] nums, int rangeL, int rangeR) {
        if (rangeL == rangeR) return true;
        if (nums[rangeL] < nums[rangeR]) return true;
        return false;
    }

    private boolean isInRange(int[] nums, int target, int rangeL, int rangeR) {
        if (nums[rangeL] <= target && target <= nums[rangeR]) return true;
        return false;
    }

    public int search(int[] nums, int target) {
        int rangeL = 0;
        int rangeR = nums.length - 1;

        while (rangeL <= rangeR) {
            int idxM = (rangeL + rangeR) / 2;

            if (rangeL == rangeR) {
                if (nums[idxM] == target) return idxM;
                else return -1;
            }
            else if (rangeL + 1 == rangeR) {
                if (nums[rangeL] == target) return rangeL;
                if (nums[rangeR] == target) return rangeR;
                return -1;
            }

            if (nums[idxM] == target) return idxM;

            if (isPartSorted(nums, rangeL, idxM - 1)) {
                if (isInRange(nums, target, rangeL, idxM - 1)) {
                    rangeR = idxM - 1;
                    continue;
                }
                else {
                    rangeL = idxM + 1;
                    continue;
                }
            }
            if (isPartSorted(nums, idxM + 1, rangeR)) {
                if (isInRange(nums, target, idxM + 1, rangeR)) {
                    rangeL = idxM + 1;
                    continue;
                }
                else {
                    rangeR = idxM - 1;
                    continue;
                }
            }

        }

        return -1;
    }
}
```

### [34. Find First and Last Position of Element in Sorted Array](https://leetcode.com/problems/find-first-and-last-position-of-element-in-sorted-array/)

```java
class Solution {
    /*
    The implementation performs a recursive-style search that explores both halves when the target is found, leading to a worst-case O(n) time complexity for arrays containing only the target value.
    */
    private int idxMin = Integer.MAX_VALUE;
    private int idxMax = Integer.MIN_VALUE;

    public int[] searchRange(int[] nums, int target) {
        Deque<Integer> ranges = new ArrayDeque<>();
        ranges.offer(0);
        ranges.offer(nums.length - 1);

        while (!ranges.isEmpty()) {
            int rangeL = ranges.poll();
            int rangeR = ranges.poll();
            if (rangeL > rangeR) continue;

            int idxM = (rangeL + rangeR) / 2;

            if (nums[idxM] == target) {
                idxMin = Math.min(idxMin, idxM);
                idxMax = Math.max(idxMax, idxM);

                // Left Part
                ranges.offer(rangeL);
                ranges.offer(idxM - 1);

                // Right Part
                ranges.offer(idxM + 1);
                ranges.offer(rangeR);
            }

            if (nums[idxM] < target) {
                ranges.offer(idxM + 1);
                ranges.offer(rangeR);
            }

            if (target < nums[idxM]) {
                ranges.offer(rangeL);
                ranges.offer(idxM - 1);
            }
        }

        if (idxMin == Integer.MAX_VALUE) return new int[]{-1, -1};
        return new int[]{idxMin, idxMax};
    }
}
```

```java
class Solution {

        LEFT, RIGHT;
    }

    private int binSearch(int[] nums, int target, Direction dir) {
        int rangeL = 0;
        int rangeR = nums.length - 1;
        int rv;
        if (dir == Direction.LEFT) rv = Integer.MAX_VALUE;
        else rv = Integer.MIN_VALUE;

                    rangeR = idxM - 1;
                }
                else {
                    rv = Math.max(rv, idxM);
                    rangeL = idxM + 1;
                }
            }
            else if (nums[idxM] < target) rangeL = idxM + 1;
            else rangeR = idxM - 1;
        }

        return rv;
    }

        int idxL = binSearch(nums, target, Direction.LEFT);
        int idxR = binSearch(nums, target, Direction.RIGHT);

        return new int[]{idxL, idxR};
    }
```

```java
                return rv;

                rangeR = idxM - 1;
            }
        }
```

### [4. Median of Two Sorted Arrays](https://leetcode.com/problems/median-of-two-sorted-arrays/)

```java
class Solution {
    public double findMedianSortedArrays(int[] nums1, int[] nums2) {
        if (nums1.length == 0 && nums2.length == 0) return -1.0;
        if (nums1.length == 0) {
            int n2 = nums2.length;
            int idxM = n2 / 2;
            return (n2 % 2 == 1) ? nums2[idxM] : (nums2[idxM - 1] + nums2[idxM]) / 2.0;
        }
        if (nums2.length == 0) {
            int n1 = nums1.length;
            int idxM = n1 / 2;
            return (n1 % 2 == 1) ? nums1[idxM] : (nums1[idxM - 1] + nums1[idxM]) / 2.0;
        }

        int rangeL = 0;
        int rangeR = nums1.length;
        int nHalf = (nums1.length + nums2.length + 1) / 2;

        while (rangeL <= rangeR) {
            int nFrom1 = (rangeL + rangeR) / 2;
            int nFrom2 = nHalf - nFrom1;
            if (nFrom2 < 0) {
                rangeR = nFrom1 - 1;
                continue;
            }
            else if (nFrom2 > nums2.length) {
                rangeL = nFrom1 + 1;
                continue;
            }

            int maxFromLeft1 = (nFrom1 == 0) ? Integer.MIN_VALUE : nums1[nFrom1 - 1];
            int minFromRight1 = (nFrom1 == nums1.length) ? Integer.MAX_VALUE : nums1[nFrom1];

            int maxFromLeft2 = (nFrom2 == 0) ? Integer.MIN_VALUE : nums2[nFrom2 - 1];
            int minFromRight2 = (nFrom2 == nums2.length) ? Integer.MAX_VALUE : nums2[nFrom2];

            if (maxFromLeft1 <= minFromRight2 && maxFromLeft2 <= minFromRight1) {
                //found
                if ((nums1.length + nums2.length) % 2 == 0) {
                    int maxLeft = Math.max(maxFromLeft1, maxFromLeft2);
                    int minRight = Math.min(minFromRight1, minFromRight2);
                    return (maxLeft + minRight) / 2.0;
                }
                return Math.max(maxFromLeft1, maxFromLeft2);
            }

            if (maxFromLeft1 > minFromRight2) rangeR = nFrom1 - 1;
            else rangeL = nFrom1 + 1;
        }

        return -1.0;
    }
}
```

```java
            }
        }

        return rv;
    }
```

### [295. Find Median from Data Stream](https://leetcode.com/problems/find-median-from-data-stream/)

```java
class MedianFinder {

    private Queue<Integer> maxHeap = new PriorityQueue<>((a, b) -> {
        return Integer.compare(b, a);
    });
    private Queue<Integer> minHeap = new PriorityQueue<>();

    public MedianFinder() {
    }

    public void addNum(int num) {
        // num's destination: left or right?
        if (maxHeap.isEmpty()) {
            maxHeap.offer(num);
            return;
        }
        if (minHeap.isEmpty()) {
            if (num < maxHeap.peek()) {
                minHeap.offer(maxHeap.poll());
                maxHeap.offer(num);
                return;
            }
            minHeap.offer(num);
            return;
        }

        if (num <= maxHeap.peek()) {
            if (maxHeap.size() == minHeap.size() + 1) minHeap.offer(maxHeap.poll());
            maxHeap.offer(num);
            return;
        }

        if (num >= minHeap.peek()) {
            if (maxHeap.size() == minHeap.size()) maxHeap.offer(minHeap.poll());
            minHeap.offer(num);
            return;
        }

        if (maxHeap.size() == minHeap.size()) maxHeap.offer(num);
        else if (maxHeap.size() == minHeap.size() + 1) minHeap.offer(num);
    }

    public double findMedian() {
        if (maxHeap.size() == minHeap.size()) return (maxHeap.peek() + minHeap.peek()) / 2.0;
        else return maxHeap.peek();
    }
}
```

### [67. Add Binary](https://leetcode.com/problems/add-binary/)

```java
class Solution {
    public String addBinary(String a, String b) {
        StringBuilder sb = new StringBuilder();

        int idxA = a.length() - 1;
        int idxB = b.length() - 1;
        int remain = 0;
        while (idxA >= 0 || idxB >= 0 || remain > 0) {
            int partSum = remain;

            if (idxA < 0 && idxB < 0) {
                sb.append(partSum);
                break;
            }

            if (idxA < 0) {
                partSum += (b.charAt(idxB) - '0');
                idxB -= 1;
            }
            else if (idxB < 0) {
                partSum += (a.charAt(idxA) - '0');
                idxA -= 1;
            }
            else {
                partSum += (a.charAt(idxA) - '0' + b.charAt(idxB) - '0');
                idxA -= 1;
                idxB -= 1;
            }

            sb.append(partSum % 2);
            remain = partSum / 2;
        }

        return sb.reverse().toString();
    }
}
```
