### [146. LRU Cache](https://leetcode.com/problems/lru-cache/)

```java
class Node {
    private int key, value;
    protected Node left, right;

    public Node(int key, int value) {
        this.key = key;
        this.value = value;
        this.left = null;
        this.right = null;
    }

    public void setValue(int value) {
        this.value = value;
    }

    public int key() {
        return key;
    }

    public int value() {
        return value;
    }

    public static void detach(Node target) {
        Node nodeL = target.left;
        Node nodeR = target.right;

        if (nodeL != null) nodeL.right = nodeR;
        if (nodeR != null) nodeR.left = nodeL;

        target.left = null;
        target.right = null;
    }

    public static void attachOnRight(Node nodeL, Node target) {
        Node nodeR = nodeL.right;

        nodeL.right = target;
        target.left = nodeL;

        if (nodeR != null) {
            target.right = nodeR;
            nodeR.left = target;
        }
    }
}

class LRUCache {
    // how can I manage the history of use in constant time??
    // history in a linked list
    // but once the value is given, detach and attach should be done in constant time
    // that means I need my own node class

    private Map<Integer, Node> nodemap;
    private int capacity, nodecnt;
    private Node headNode, tailNode;

    public LRUCache(int capacity) {
        this.nodemap = new HashMap<>();
        this.capacity = capacity;
        this.nodecnt = 0;
    }

    public void updateTail(Node target) {
        if (target == tailNode) return;
        if (target == headNode) {
            headNode = headNode.right;
        }

        Node.detach(target);
        Node.attachOnRight(tailNode, target);
        tailNode = target;
    }

    public int get(int key) {
        if (!nodemap.containsKey(key)) return -1;

        Node touched = nodemap.get(key);
        updateTail(touched);

        return touched.value();
    }

    public void removeLRU() {
        // remove from the list
        Node target = headNode;
        Node nextHead = headNode.right;

        Node.detach(target);
        headNode = nextHead;

        // remove from the hashmap
        int removeKey = target.key();
        nodemap.remove(removeKey);
    }

    public void put(int key, int value) {
        if (nodemap.containsKey(key)) {
            Node target = nodemap.get(key);
            target.setValue(value);
            updateTail(target);
        }
        else {
            Node newNode = new Node(key, value);
            nodemap.put(key, newNode);

            if (nodecnt == capacity) {
                removeLRU();
                if (capacity == 1) {
                    headNode = newNode;
                    tailNode = newNode;
                }
            }
            else {
                if (nodecnt == 0) {
                    headNode = newNode;
                    tailNode = newNode;
                }
                nodecnt++;
            }

            updateTail(newNode);
        }
    }
}

/**
 * Your LRUCache object will be instantiated and called as such:
 * LRUCache obj = new LRUCache(capacity);
 * int param_1 = obj.get(key);
 * obj.put(key,value);
 */
```

### [104. Maximum Depth of Binary Tree](https://leetcode.com/problems/maximum-depth-of-binary-tree/)

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public int maxDepth(TreeNode root) {
        if (root == null) return 0;
        return Math.max(maxDepth(root.left), maxDepth(root.right)) + 1;
    }
}
```

### [226. Invert Binary Tree](https://leetcode.com/problems/invert-binary-tree/)

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public TreeNode invertTree(TreeNode root) {
        if (root == null) return null;

        TreeNode nodeL = root.left;
        TreeNode nodeR = root.right;
        root.left = nodeR;
        root.right = nodeL;

        invertTree(root.left);
        invertTree(root.right);

        return root;
    }
}
```

```java
/**

 *     TreeNode right;

 *     TreeNode(int val) { this.val = val; }

 *         this.right = right;
 *     }
 * }
 */
class Solution {
    private boolean _isSymmetric(TreeNode nodeL, TreeNode nodeR) {
        if (nodeL == null && nodeR == null) return true;
        if (nodeL == null || nodeR == null || nodeL.val != nodeR.val) return false;

        return _isSymmetric(root.left, root.right);
    }
```

### [105. Construct Binary Tree from Preorder and Inorder Traversal](https://leetcode.com/problems/construct-binary-tree-from-preorder-and-inorder-traversal/)

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    Map<Integer, Integer> pIdxMap, iIdxMap;
    int[] preorder, inorder;

    /*
    ## PREORDER: [ root      +      +      +      +      [+]      -      -      -      - ]
    - pLeftOpeningIdx
    - pLeftClosingIdx
    - pRightOpeningIdx
    - pRightClosingIdx

    ## INORDER:  [ +      +      +      +      [+]      root      -      -      -      - ]
    - iLeftOpeningIdx
    - iLeftClosingIdx
    - iRightOpeningIdx
    - iRightClosingIdx
    */

    private TreeNode _bt(int pOpeningIdx, int pClosingIdx, int iOpeningIdx, int iClosingIdx) {
        if (pOpeningIdx >= pClosingIdx) return null;
        if (iOpeningIdx >= iClosingIdx) return null;

        int pLeftOpeningIdx, pLeftClosingIdx, pRightOpeningIdx, pRightClosingIdx;
        int iLeftOpeningIdx, iLeftClosingIdx, iRightOpeningIdx, iRightClosingIdx;

        int rootValue = preorder[pOpeningIdx];
        TreeNode rootNode = new TreeNode(rootValue);

        iLeftOpeningIdx = iOpeningIdx;
        iLeftClosingIdx = iIdxMap.get(rootValue);
        iRightOpeningIdx = iLeftClosingIdx + 1;
        iRightClosingIdx = iClosingIdx;

        pLeftOpeningIdx = pOpeningIdx + 1;
        pLeftClosingIdx = pLeftOpeningIdx + iLeftClosingIdx - iOpeningIdx;
        pRightOpeningIdx = pLeftClosingIdx;
        pRightClosingIdx = pClosingIdx;

        if (iLeftOpeningIdx != iLeftClosingIdx) rootNode.left = _bt(pLeftOpeningIdx, pLeftClosingIdx, iLeftOpeningIdx, iLeftClosingIdx);
        else rootNode.left = null;
        rootNode.right = _bt(pRightOpeningIdx, pRightClosingIdx, iRightOpeningIdx, iRightClosingIdx);

        return rootNode;
    }

    public TreeNode buildTree(int[] preorder, int[] inorder) {
        this.preorder = preorder;
        this.inorder = inorder;

        int n = preorder.length;

        pIdxMap = new HashMap<>();
        iIdxMap = new HashMap<>();
        for (int idx = 0; idx < n; idx++) {
            pIdxMap.put(preorder[idx], idx);
            iIdxMap.put(inorder[idx], idx);
        }

        return _bt(0, n, 0, n);
    }
}
```

### [106. Construct Binary Tree from Inorder and Postorder Traversal](https://leetcode.com/problems/construct-binary-tree-from-inorder-and-postorder-traversal/)

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    Map<Integer, Integer> iIdxMap;
    int[] inorder, postorder;
    int pIdx;

    private TreeNode _bt(int openingIdx, int closingIdx) {
        if (openingIdx > closingIdx) return null;
        if (pIdx < 0) return null;

        int rootValue = postorder[pIdx--];
        TreeNode rootNode = new TreeNode(rootValue);

        // subtrees
        int rootIdx = iIdxMap.get(rootValue);
        TreeNode treeR = _bt(rootIdx + 1, closingIdx);
        TreeNode treeL = _bt(openingIdx, rootIdx - 1);

        rootNode.right = treeR;
        rootNode.left = treeL;

        return rootNode;
    }

    public TreeNode buildTree(int[] inorder, int[] postorder) {
        this.inorder = inorder;
        this.postorder = postorder;

        iIdxMap = new HashMap<>();
        int n = inorder.length;
        for (int idx = 0; idx < n; idx++) {
            iIdxMap.put(inorder[idx], idx);
        }

        pIdx = n - 1;
        return _bt(0, n - 1);
    }
}
```

### [117. Populating Next Right Pointers in Each Node II](https://leetcode.com/problems/populating-next-right-pointers-in-each-node-ii/)

```java
/*
// Definition for a Node.
class Node {
    public int val;
    public Node left;
    public Node right;
    public Node next;

    public Node() {}

    public Node(int _val) {
        val = _val;
    }

    public Node(int _val, Node _left, Node _right, Node _next) {
        val = _val;
        left = _left;
        right = _right;
        next = _next;
    }
};
*/

class Solution {
    public Node connect(Node root) {
        Node upNode = root;

        while (upNode != null) { // vertical
            Node downLeader = new Node(-1);
            Node downAppendingPoint = downLeader;

            while (upNode != null) { // horizontal
                if (upNode.left != null) {
                    downAppendingPoint.next = upNode.left;
                    downAppendingPoint = downAppendingPoint.next;
                }
                if (upNode.right != null) {
                    downAppendingPoint.next = upNode.right;
                    downAppendingPoint = downAppendingPoint.next;
                }
                upNode = upNode.next;
            }

            upNode = downLeader.next;
        }

        return root;
    }
}
```
