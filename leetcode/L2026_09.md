### [146. LRU Cache](https://leetcode.com/problems/lru-cache/)

```java
class Node {
    private int key, value;
    protected Node left, right;

    public Node(int key, int value) {
        this.key = key;
        this.value = value;
        this.left = null;
        this.right = null;
    }

    public void setValue(int value) {
        this.value = value;
    }

    public int key() {
        return key;
    }

    public int value() {
        return value;
    }

    public static void detach(Node target) {
        Node nodeL = target.left;
        Node nodeR = target.right;

        if (nodeL != null) nodeL.right = nodeR;
        if (nodeR != null) nodeR.left = nodeL;

        target.left = null;
        target.right = null;
    }

    public static void attachOnRight(Node nodeL, Node target) {
        Node nodeR = nodeL.right;

        nodeL.right = target;
        target.left = nodeL;

        if (nodeR != null) {
            target.right = nodeR;
            nodeR.left = target;
        }
    }
}

class LRUCache {
    // how can I manage the history of use in constant time??
    // history in a linked list
    // but once the value is given, detach and attach should be done in constant time
    // that means I need my own node class

    private Map<Integer, Node> nodemap;
    private int capacity, nodecnt;
    private Node headNode, tailNode;

    public LRUCache(int capacity) {
        this.nodemap = new HashMap<>();
        this.capacity = capacity;
        this.nodecnt = 0;
    }

    public void updateTail(Node target) {
        if (target == tailNode) return;
        if (target == headNode) {
            headNode = headNode.right;
        }

        Node.detach(target);
        Node.attachOnRight(tailNode, target);
        tailNode = target;
    }

    public int get(int key) {
        if (!nodemap.containsKey(key)) return -1;

        Node touched = nodemap.get(key);
        updateTail(touched);

        return touched.value();
    }

    public void removeLRU() {
        // remove from the list
        Node target = headNode;
        Node nextHead = headNode.right;

        Node.detach(target);
        headNode = nextHead;

        // remove from the hashmap
        int removeKey = target.key();
        nodemap.remove(removeKey);
    }

    public void put(int key, int value) {
        if (nodemap.containsKey(key)) {
            Node target = nodemap.get(key);
            target.setValue(value);
            updateTail(target);
        }
        else {
            Node newNode = new Node(key, value);
            nodemap.put(key, newNode);

            if (nodecnt == capacity) {
                removeLRU();
                if (capacity == 1) {
                    headNode = newNode;
                    tailNode = newNode;
                }
            }
            else {
                if (nodecnt == 0) {
                    headNode = newNode;
                    tailNode = newNode;
                }
                nodecnt++;
            }

            updateTail(newNode);
        }
    }
}

 * LRUCache obj = new LRUCache(capacity);
 * int param_1 = obj.get(key);
 * obj.put(key,value);
 */
```

### [105. Construct Binary Tree from Preorder and Inorder Traversal](https://leetcode.com/problems/construct-binary-tree-from-preorder-and-inorder-traversal/)

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    Map<Integer, Integer> pIdxMap, iIdxMap;
    int[] preorder, inorder;

    /*
    ## PREORDER: [ root      +      +      +      +      [+]      -      -      -      - ]
    - pLeftOpeningIdx
    - pLeftClosingIdx
    - pRightOpeningIdx
    - pRightClosingIdx

    ## INORDER:  [ +      +      +      +      [+]      root      -      -      -      - ]
    - iLeftOpeningIdx
    - iLeftClosingIdx
    - iRightOpeningIdx
    - iRightClosingIdx
    */

    private TreeNode _bt(int pOpeningIdx, int pClosingIdx, int iOpeningIdx, int iClosingIdx) {
        if (pOpeningIdx >= pClosingIdx) return null;
        if (iOpeningIdx >= iClosingIdx) return null;

        int pLeftOpeningIdx, pLeftClosingIdx, pRightOpeningIdx, pRightClosingIdx;
        int iLeftOpeningIdx, iLeftClosingIdx, iRightOpeningIdx, iRightClosingIdx;

        int rootValue = preorder[pOpeningIdx];
        TreeNode rootNode = new TreeNode(rootValue);

        iLeftOpeningIdx = iOpeningIdx;
        iLeftClosingIdx = iIdxMap.get(rootValue);
        iRightOpeningIdx = iLeftClosingIdx + 1;
        iRightClosingIdx = iClosingIdx;

        pLeftOpeningIdx = pOpeningIdx + 1;
        pLeftClosingIdx = pLeftOpeningIdx + iLeftClosingIdx - iOpeningIdx;
        pRightOpeningIdx = pLeftClosingIdx;
        pRightClosingIdx = pClosingIdx;

        if (iLeftOpeningIdx != iLeftClosingIdx) rootNode.left = _bt(pLeftOpeningIdx, pLeftClosingIdx, iLeftOpeningIdx, iLeftClosingIdx);
        else rootNode.left = null;
        rootNode.right = _bt(pRightOpeningIdx, pRightClosingIdx, iRightOpeningIdx, iRightClosingIdx);

        return rootNode;
    }

    public TreeNode buildTree(int[] preorder, int[] inorder) {
        this.preorder = preorder;
        this.inorder = inorder;

        int n = preorder.length;

        pIdxMap = new HashMap<>();
        iIdxMap = new HashMap<>();
        for (int idx = 0; idx < n; idx++) {
            pIdxMap.put(preorder[idx], idx);
            iIdxMap.put(inorder[idx], idx);
        }

        return _bt(0, n, 0, n);
    }
}
```

```java
    int[] inorder, postorder;
    int pIdx;
```

```java
            upNode = downLeader.next;
```

```java
        TreeNode ptr = root;
        while (true) {

        lastL.right = root.right;
        root.right = root.left;
        root.left = null;

        return;
    }
```

### [236. Lowest Common Ancestor of a Binary Tree](https://leetcode.com/problems/lowest-common-ancestor-of-a-binary-tree/)

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode(int x) { val = x; }
 * }
 */
class Solution {
    // can record parent of each node using hash
    // can trace back the ancestors to find common one
    // make list of ancestors of one node
    // then traverse the other one's ancestor nodes until we find duplicate in the lists
    private Map<TreeNode, TreeNode> parentMap;
    private Set<TreeNode> ancestorSetOfFirstNode;

    private void _spm(TreeNode root) {
        if (root == null) return;

        if (root.left != null) {
            parentMap.put(root.left, root);
            _spm(root.left);
        }
        if (root.right != null) {
            parentMap.put(root.right, root);
            _spm(root.right);
        }

        return;
    }

    private void setParentMap(TreeNode root) {
        parentMap = new HashMap<>();
        _spm(root);
        return;
    }

    private void setAncestorSetOfFirstNode(TreeNode childNode) {
        ancestorSetOfFirstNode = new HashSet<>();
        ancestorSetOfFirstNode.add(childNode);

        while (parentMap.containsKey(childNode)) {
            TreeNode parentNode = parentMap.get(childNode);
            ancestorSetOfFirstNode.add(parentNode);
            childNode = parentNode;
        }

        return;
    }

    private TreeNode findFirstCommon(TreeNode childNode) {
        if (ancestorSetOfFirstNode.contains(childNode)) {
            return childNode;
        }

        while (parentMap.containsKey(childNode)) {
            TreeNode parentNode = parentMap.get(childNode);
            if (ancestorSetOfFirstNode.contains(parentNode)) {
                return parentNode;
            }
            childNode = parentNode;
        }

        return null;
    }

    public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) {
        if (root == null || p == null || q == null) {
            return null;
        }

        setParentMap(root);
        setAncestorSetOfFirstNode(p);
        return findFirstCommon(q);
    }
}
```

```java
    private void _edge(char[][] board, int r, int c) {
        if (!(0 <= r && r < m)) return;
        if (!(0 <= c && c < n)) return;

        if (board[r][c] == 'X') return;
        if (board[r][c] == 'e') return;

        board[r][c] = 'e';

        _edge(board, r - 1, c);
        _edge(board, r + 1, c);
        _edge(board, r, c - 1);
        _edge(board, r, c + 1);
    }

    private void processEdgeRegions(char[][] board) {
        // mark edges
        for (int r = 0; r < m; r++) {
            if (board[r][0] == 'O') _edge(board, r, 0);
            if (board[r][n - 1] == 'O') _edge(board, r, n - 1);
        }
        for (int c = 0; c < n; c++) {
            if (board[0][c] == 'O') _edge(board, 0, c);
            if (board[m - 1][c] == 'O') _edge(board, m - 1, c);
        }
    }

    private void recoverEdgeRegion(char[][] board) {
        for (int r = 0; r < m; r++) {
            for (int c = 0; c < n; c++) {
                if (board[r][c] == 'e') {
                    board[r][c] = 'O';
                }
            }
        }
    }

    private void _target(char[][] board, int r, int c) {
        if (!(0 <= r && r < m)) return;
        if (!(0 <= c && c < n)) return;

        if (board[r][c] == 'X') return;
        if (board[r][c] == 'e') return;

        board[r][c] = 'X';

        _target(board, r - 1, c);
        _target(board, r + 1, c);
        _target(board, r, c - 1);
        _target(board, r, c + 1);
    }

    private void processTarget(char[][] board) {
        for (int r = 0; r < m; r++) {
            for (int c = 0; c < n; c++) {
                if (board[r][c] == 'O') {
                    _target(board, r, c);
                }
            }
        }
    }

        n = board[0].length;

        // find edge region's points and make a record of those points

        // then traverse the points to alter outside of hashmap.
```

```java
class Solution {
    private Map<String, Map<String, Double>> ratioMap;

    private void setRatioMap(List<List<String>> equations, double[] values) {
        ratioMap = new HashMap<>();

        int n = equations.size();
        for (int idx = 0; idx < n; idx++) {
            List<String> division = equations.get(idx);
            String term1 = division.get(0);
            String term2 = division.get(1);

            if (!ratioMap.containsKey(term1)) {
                Map<String, Double> innerMap = new HashMap<>();
                innerMap.put(term2, values[idx]);
                ratioMap.put(term1, innerMap);
            }
            else {
                Map<String, Double> innerMap = ratioMap.get(term1);
                innerMap.put(term2, values[idx]);
            }

            if (!ratioMap.containsKey(term2)) {
                Map<String, Double> innerMap = new HashMap<>();
                innerMap.put(term1, 1.0 / values[idx]);
                ratioMap.put(term2, innerMap);
            }
            else {
                Map<String, Double> innerMap = ratioMap.get(term2);
                innerMap.put(term1, 1.0 / values[idx]);
            }
        }

        return;
    }

    private double computeQuery(String openingNode, String closingNode, Set<String> visited) {
        visited.add(openingNode);

        Map<String, Double> innerMap = ratioMap.get(openingNode);
        if (innerMap == null) return -1.0;

        Set<String> connectedNodeSet = innerMap.keySet();

        if (connectedNodeSet.contains(closingNode)) {
            return innerMap.get(closingNode);
        }
        for (String connectingNode: connectedNodeSet) {
            if (visited.contains(connectingNode)) continue;

            double firstHalf = innerMap.get(connectingNode);
            double secondHalf = computeQuery(connectingNode, closingNode, visited);

            if (secondHalf == -1.0) continue;
            return firstHalf * secondHalf;
        }

        visited.remove(openingNode);
        return -1.0;
    }

    public double[] calcEquation(List<List<String>> equations, double[] values, List<List<String>> queries) {
        setRatioMap(equations, values);

        int n = queries.size();
        double[] rv = new double[n];
        for (int idx = 0; idx < n; idx++) {
            List<String> q = queries.get(idx);
            rv[idx] = computeQuery(q.get(0), q.get(1), new HashSet<>());
        }

        return rv;
    }
```

```java
        Deque<Integer> finishedQueue = new ArrayDeque<>();
```

```java
class Solution {
    private int[][] board;
    private Deque<Integer> searchQueue;
    private Deque<Integer> countQueue;
    private int rSize, cSize;
    private int destination;
    private int[] rv;

    private int getCellValue(int cellNumber) {
        if (cellNumber > destination) return 0;
        if (cellNumber < 1) return 0;

        int rOffset = (cellNumber - 1) / cSize + 1;
        int cOffset = (cellNumber - 1) % cSize + 1;

        int r = rSize - rOffset;
        int c = cSize - cOffset;
        if (rOffset % 2 == 1) {
            c = cOffset - 1;
        }

        return board[r][c];
    }

    private void rollDice(int cellNumber, int moveCount) {
        if (cellNumber < 1) return;
        if (cellNumber > destination) return;
        if (moveCount >= rv[cellNumber]) return;
        rv[cellNumber] = moveCount;

        for (int d = 6; d > 0; d--) {
            int v = getCellValue(cellNumber + d);

            if (v == 0) continue;
            if (v == -1) {
                searchQueue.offer(cellNumber + d);
            }
            else {
                searchQueue.offer(v);
            }
            countQueue.offer(moveCount + 1);
        }

        return;
    }

    public int snakesAndLadders(int[][] board) {
        this.board = board;
        searchQueue = new ArrayDeque<>();
        countQueue = new ArrayDeque<>();
        rSize = board.length;
        cSize = board[0].length;
        destination = rSize * cSize;
        rv = new int[destination + 1];
        Arrays.fill(rv, Integer.MAX_VALUE);

        // compute
        searchQueue.offer(1);
        countQueue.offer(0);
        while (!searchQueue.isEmpty()) {
            int cellNumber = searchQueue.poll();
            int moveCount = countQueue.poll();
            rollDice(cellNumber, moveCount);
        }

        if (rv[destination] == Integer.MAX_VALUE) return -1;
```

```java
        Set<String> mSet = new HashSet<>();

        for (String candidate: bankSet) {
            int nDiff = 0;
            for (int idx = 0; idx < gene.length(); idx++) {
                if (gene.charAt(idx) != candidate.charAt(idx)) {
                    nDiff += 1;
                }
                if (nDiff > 1) break;
            }

            if (nDiff == 1) {
                mSet.add(candidate);
            }
        }

        return mSet;
    }

    private void mutate(String gene, int mCount) {
        if (rv.getOrDefault(gene, Integer.MAX_VALUE) <= mCount) return;
        rv.put(gene, mCount);
        if (gene.equals(destination)) return;

        // possible mutations into the queue (and the following counts)
        Set<String> mutationSet = getMutationSet(gene);
        for (String nextGene: mutationSet) {
            searchQueue.offer(nextGene);
            countQueue.offer(mCount + 1);
        }

        return;
    }

    public int minMutation(String startGene, String endGene, String[] bank) {
        setBank(bank);
        searchQueue = new ArrayDeque<>();
        countQueue = new ArrayDeque<>();
        rv = new HashMap<>();
        destination = endGene;

        // compute
        searchQueue.offer(startGene);
        countQueue.offer(0);
        while (!searchQueue.isEmpty()) {
            String gene = searchQueue.poll();
            int mCount = countQueue.poll();
            mutate(gene, mCount);
        }

        return rv.getOrDefault(endGene, -1);
    }
```

### [127. Word Ladder](https://leetcode.com/problems/word-ladder/)

```java
class Solution {
    private Set<String> bankSet;
    private Map<String, Integer> countMap;
    private String destination;
    private Deque<String> pendingQueue;
    private Deque<Integer> countQueue;

    private void setBank(List<String> wordList) {
        if (wordList == null) return;

        bankSet = new HashSet<>();

        int n = wordList.size();
        for (int idx = 0; idx < n; idx++) {
            bankSet.add(wordList.get(idx));
        }

        return;
    }

    private Set<String> getNextSet(String currentWord) {
        Set<String> nSet = new HashSet<>();

        for (String cand: bankSet) {
            int nDiff = 0;
            for (int idx = 0; idx < cand.length(); idx++) {
                if (currentWord.charAt(idx) != cand.charAt(idx)) {
                    nDiff += 1;
                }
                if (nDiff > 1) continue;
            }
            if (nDiff == 1) {
                nSet.add(cand);
            }
        }

        return nSet;
    }

    private void mutate(String currentWord, int mCount) {
        if (countMap.getOrDefault(currentWord, Integer.MAX_VALUE) <= mCount) return;
        countMap.put(currentWord, mCount);

        if (currentWord.equals(destination)) return;

        // possible next steps into queue and accordingly counts
        Set<String> nextSet = getNextSet(currentWord);
        for (String nextItem: nextSet) {
            pendingQueue.offer(nextItem);
            countQueue.offer(mCount + 1);
        }

        return;
    }

    public int ladderLength(String beginWord, String endWord, List<String> wordList) {
        if (beginWord.equals(endWord)) return 1;
        if (wordList == null) return 0;
        if (wordList.size() == 0) return 0;

        setBank(wordList);
        countMap = new HashMap<>();
        destination = endWord;
        pendingQueue = new ArrayDeque<>();
        countQueue = new ArrayDeque<>();

        // compute
        pendingQueue.add(beginWord);
        countQueue.add(1);
        while (!pendingQueue.isEmpty()) {
            String item = pendingQueue.poll();
            int mCount = countQueue.poll();
            mutate(item, mCount);
        }

        return countMap.getOrDefault(destination, 0);
    }
}
```

### [208. Implement Trie (Prefix Tree)](https://leetcode.com/problems/implement-trie-prefix-tree/)

```java
class Trie {
    private static class Node {
        Node[] edgeList;
        boolean isEnd;

        public Node() {
            edgeList = new Node['z' - 'a' + 1];
            isEnd = false;
        }
    }

    private Node rootNode;

    public Trie() {
        rootNode = new Node();
    }

    public void insert(String word) {
        int n = word.length();

        Node traversingNode = rootNode;
        for (int idx = 0; idx < n; idx++) {
            char letter = word.charAt(idx);

            Node[] edgeList = traversingNode.edgeList;
            if (edgeList[letter - 'a'] == null) {
                edgeList[letter - 'a'] = new Node();
            }

            traversingNode = edgeList[letter - 'a'];
        }

        traversingNode.isEnd = true;

        return;
    }

    private Node tracePrefix(String prefix) {
        int n = prefix.length();

        Node traversingNode = rootNode;
        for (int idx = 0; idx < n; idx++) {
            char letter = prefix.charAt(idx);
            Node[] edgeList = traversingNode.edgeList;

            if (edgeList[letter - 'a'] == null) {
                return null;
            }

            traversingNode = edgeList[letter - 'a'];
        }

        return traversingNode;
    }

    public boolean search(String word) {
        Node node = tracePrefix(word);

        if (node == null) return false;
        if (node.isEnd) return true;
        return false;
    }

    public boolean startsWith(String prefix) {
        Node node = tracePrefix(prefix);

        if (node == null) return false;
        return true;
    }
}
```

### [211. Design Add and Search Words Data Structure](https://leetcode.com/problems/design-add-and-search-words-data-structure/)

```java
class WordDictionary {
    private static class TrieNode {
        TrieNode[] edge;
        boolean isEnd;

        TrieNode() {
            edge = new TrieNode['z' - 'a' + 1];
            isEnd = false;
        }
    }

    private TrieNode rootNode;

    public WordDictionary() {
        rootNode = new TrieNode();
    }

    public void addWord(String word) {
        int n = word.length();

        TrieNode traversingNode = rootNode;
        for (int idx = 0; idx < n; idx++) {
            char letter = word.charAt(idx);
            int letterIdx = letter - 'a';
            TrieNode[] edge = traversingNode.edge;

            if (edge[letterIdx] == null) {
                edge[letterIdx] = new TrieNode();
            }

            traversingNode = edge[letterIdx];
        }
        traversingNode.isEnd = true;

        return;
    }

    public boolean search(String word) {
        int n = word.length();

        Deque<TrieNode> nodeQueue = new ArrayDeque<>();
        Deque<Integer> idxQueue = new ArrayDeque<>();
        nodeQueue.offer(rootNode);
        idxQueue.offer(0);

        while (!nodeQueue.isEmpty()) {
            TrieNode node = nodeQueue.poll();
            TrieNode[] edge = node.edge;
            int idx = idxQueue.poll();
            char letter = word.charAt(idx);

            if (letter == '.') {
                if (idx == n - 1) {
                    for (char c = 'a'; c <= 'z'; c++) {
                        if (edge[c - 'a'] != null && edge[c - 'a'].isEnd) return true;
                    }

                    continue;
                }
                for (char c = 'a'; c <= 'z'; c++) {
                    if (edge[c - 'a'] != null) {
                        nodeQueue.offer(edge[c - 'a']);
                        idxQueue.offer(idx + 1);
                    }
                }
                continue;
            }

            if (edge[letter - 'a'] == null) {
                continue;
            }
            if (idx == n - 1) {
                if (edge[letter - 'a'].isEnd) return true;
                continue;
            }
            nodeQueue.offer(edge[letter - 'a']);
            idxQueue.offer(idx + 1);
        }

        return false;
    }
}
```

### [212. Word Search II](https://leetcode.com/problems/word-search-ii/)

```java
class Solution {
    private List<String> rv;
    private char[][] board;
    private boolean[][] visited;
    private int rsize, csize;
    private Map<Character, Set<int[]>> openMap;

    private boolean recursiveFind(String target, int idx, int r, int c) {
        int n = target.length();
        if (idx >= n) return false;
        if (r < 0 || r >= rsize) return false;
        if (c < 0 || c >= csize) return false;
        if (visited[r][c]) return false;
        if (board[r][c] != target.charAt(idx)) return false;

        // exit condition
        if (idx == n - 1) {
            if (board[r][c] == target.charAt(idx)) return true;
            else return false;
        }

        visited[r][c] = true;

        // bfs
        int[][] direction = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
        for (int[] d: direction) {
            if ( recursiveFind(target, idx + 1, r + d[0], c + d[1])) {
                return true;
            }
        }

        visited[r][c] = false;
        return false;
    }

    private void processWord(String word) {
        if (!openMap.containsKey(word.charAt(0))) return;

        Set<int[]> openingPoints = openMap.get(word.charAt(0));
        for (int[] point: openingPoints) {
            int r = point[0];
            int c = point[1];

            visited = new boolean[rsize][csize];

            if (recursiveFind(word, 0, r, c)) {
                rv.add(word);
                return;
            }
        }

        return;
    }

    private void setOpenMap() {
        openMap = new HashMap<>();

        for (int r = 0; r < rsize; r++) {
            for (int c = 0; c < csize; c++) {
                Set<int[]> innerset;
                if (openMap.containsKey(board[r][c])) {
                    innerset = openMap.get(board[r][c]);
                }
                else {
                    innerset = new HashSet<>();
                    openMap.put(board[r][c], innerset);
                }

                int[] point = {r, c};
                innerset.add(point);
            }
        }

        return;
    }

    public List<String> findWords(char[][] board, String[] words) {
        this.board = board;
        rsize = board.length;
        csize = board[0].length;
        visited = new boolean[rsize][csize];

        setOpenMap();

        rv = new ArrayList<>();
        for (String word: words) {
            processWord(word);
        }

        return rv;
    }
}
```

### [212. Word Search II](https://leetcode.com/problems/word-search-ii/)

```java
class Solution {
    private class TrieNode {
        TrieNode[] edge;
        String word;

        TrieNode() {
            edge = new TrieNode['z' - 'a' + 1];
            word = null;
        }
    }

    private TrieNode rootNode = new TrieNode();
    private List<String> rv = new ArrayList<>();
    private char[][] board;

    private void setTrie(String word) {
        TrieNode node = rootNode;
        int n = word.length();

        for (int idx = 0; idx < n; idx++) {
            if (node.edge[word.charAt(idx) - 'a'] == null) {
                node.edge[word.charAt(idx) - 'a'] = new TrieNode();
            }

            node = node.edge[word.charAt(idx) - 'a'];
        }

        node.word = word;
        return;
    }

    private void dfs(int r, int c, TrieNode node) {
        if (r < 0 || r >= board.length) return;
        if (c < 0 || c >= board[0].length) return;
        if (board[r][c] == '_') return;
        if (node.edge[board[r][c] - 'a'] == null) return;

        char letter = board[r][c];
        board[r][c] = '_';
        TrieNode linkedNode = node.edge[letter - 'a'];
        if (linkedNode.word != null) {
            rv.add(linkedNode.word);
            linkedNode.word = null;
        }

        int[][] directions = {{1, 0}, {0, 1}, {-1, 0}, {0, -1}};
        for (int[] d: directions) {
            dfs(r + d[0], c + d[1], linkedNode);
        }

        board[r][c] = letter;
        return;
    }

    public List<String> findWords(char[][] board, String[] words) {
        this.board = board;

        for (String word: words) {
            setTrie(word);
        }

        for (int r = 0; r < board.length; r++) {
            for (int c = 0; c < board[0].length; c++) {
                dfs(r, c, rootNode);
            }
        }

        return rv;
    }
}
```

### [17. Letter Combinations of a Phone Number](https://leetcode.com/problems/letter-combinations-of-a-phone-number/)

```java
class Solution {
    private List<String> rv = new ArrayList<>();
    private Map<Integer, String> letterMap = new HashMap<>();
    private StringBuilder sb = new StringBuilder();

    private void setLetterMap() {
        letterMap.put(2, "abc");
        letterMap.put(3, "def");
        letterMap.put(4, "ghi");
        letterMap.put(5, "jkl");
        letterMap.put(6, "mno");
        letterMap.put(7, "pqrs");
        letterMap.put(8, "tuv");
        letterMap.put(9, "wxyz");
    }

    private void dfs(String digits, int idx) {
        if (idx == digits.length()) {
            rv.add(sb.toString());
            return;
        }

        String letterList = letterMap.get(digits.charAt(idx) - '0');
        for (char letter: letterList.toCharArray()) {
            sb.append(letter);
            dfs(digits, idx + 1);
            sb.deleteCharAt(sb.length() - 1);
        }

        return;
    }

    public List<String> letterCombinations(String digits) {
        setLetterMap();
        dfs(digits, 0);
        return rv;
    }
}
```
