### [146. LRU Cache](https://leetcode.com/problems/lru-cache/)

```java
class Node {
    private int key, value;
    protected Node left, right;

    public Node(int key, int value) {
        this.key = key;
        this.value = value;
        this.left = null;
        this.right = null;
    }

    public void setValue(int value) {
        this.value = value;
    }

    public int key() {
        return key;
    }

    public int value() {
        return value;
    }

    public static void detach(Node target) {
        Node nodeL = target.left;
        Node nodeR = target.right;

        if (nodeL != null) nodeL.right = nodeR;
        if (nodeR != null) nodeR.left = nodeL;

        target.left = null;
        target.right = null;
    }

    public static void attachOnRight(Node nodeL, Node target) {
        Node nodeR = nodeL.right;

        nodeL.right = target;
        target.left = nodeL;

        if (nodeR != null) {
            target.right = nodeR;
            nodeR.left = target;
        }
    }
}

class LRUCache {
    // how can I manage the history of use in constant time??
    // history in a linked list
    // but once the value is given, detach and attach should be done in constant time
    // that means I need my own node class

    private Map<Integer, Node> nodemap;
    private int capacity, nodecnt;
    private Node headNode, tailNode;

    public LRUCache(int capacity) {
        this.nodemap = new HashMap<>();
        this.capacity = capacity;
        this.nodecnt = 0;
    }

    public void updateTail(Node target) {
        if (target == tailNode) return;
        if (target == headNode) {
            headNode = headNode.right;
        }

        Node.detach(target);
        Node.attachOnRight(tailNode, target);
        tailNode = target;
    }

    public int get(int key) {
        if (!nodemap.containsKey(key)) return -1;

        Node touched = nodemap.get(key);
        updateTail(touched);

        return touched.value();
    }

    public void removeLRU() {
        // remove from the list
        Node target = headNode;
        Node nextHead = headNode.right;

        Node.detach(target);
        headNode = nextHead;

        // remove from the hashmap
        int removeKey = target.key();
        nodemap.remove(removeKey);
    }

    public void put(int key, int value) {
        if (nodemap.containsKey(key)) {
            Node target = nodemap.get(key);
            target.setValue(value);
            updateTail(target);
        }
        else {
            Node newNode = new Node(key, value);
            nodemap.put(key, newNode);

            if (nodecnt == capacity) {
                removeLRU();
                if (capacity == 1) {
                    headNode = newNode;
                    tailNode = newNode;
                }
            }
            else {
                if (nodecnt == 0) {
                    headNode = newNode;
                    tailNode = newNode;
                }
                nodecnt++;
            }

            updateTail(newNode);
        }
    }
}

/**
 * Your LRUCache object will be instantiated and called as such:
 * LRUCache obj = new LRUCache(capacity);
 * int param_1 = obj.get(key);
 * obj.put(key,value);
 */
```

### [104. Maximum Depth of Binary Tree](https://leetcode.com/problems/maximum-depth-of-binary-tree/)

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public int maxDepth(TreeNode root) {
        if (root == null) return 0;
        return Math.max(maxDepth(root.left), maxDepth(root.right)) + 1;
    }
}
```

### [100. Same Tree](https://leetcode.com/problems/same-tree/)

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public boolean isSameTree(TreeNode p, TreeNode q) {
        if (p == null && q == null) return true;
        if (p == null || q == null || p.val != q.val) return false;
        return isSameTree(p.left, q.left) && isSameTree(p.right, q.right);
    }
}
```

### [226. Invert Binary Tree](https://leetcode.com/problems/invert-binary-tree/)

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public TreeNode invertTree(TreeNode root) {
        if (root == null) return null;

        TreeNode nodeL = root.left;
        TreeNode nodeR = root.right;
        root.left = nodeR;
        root.right = nodeL;

        invertTree(root.left);
        invertTree(root.right);

        return root;
    }
}
```

### [101. Symmetric Tree](https://leetcode.com/problems/symmetric-tree/)

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    private boolean _isSymmetric(TreeNode nodeL, TreeNode nodeR) {
        if (nodeL == null && nodeR == null) return true;
        if (nodeL == null || nodeR == null || nodeL.val != nodeR.val) return false;
        return _isSymmetric(nodeL.left, nodeR.right) && _isSymmetric(nodeL.right, nodeR.left);
    }

    public boolean isSymmetric(TreeNode root) {
        return _isSymmetric(root.left, root.right);
    }
}
```
