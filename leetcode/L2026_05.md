### [LeetCode 70. Climbing Stairs](https://leetcode.com/problems/climbing-stairs/)

```python
class Solution:
    def climbStairs(self, n: int) -> int:
        if n < 2:
            return 1

        tmp1 = tmp2 = 1
        for idx in range(2, n):
            v_idx = tmp1 + tmp2
            tmp1 = tmp2
            tmp2 = v_idx

        return tmp1 + tmp2
```

### [118. Pascal's Triangle](https://leetcode.com/problems/pascals-triangle/)

```python
class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        if numRows == 1:
            return [[1]]
        elif numRows == 2:
            return[[1], [1,1]]

        answer = [[1], [1,1]]
        remember = [1,1]
        for r_idx in range(2, numRows):
            # compute row[r_idx]
            tmp = [1]
            for c_idx in range(1, r_idx):
                tmp.append(remember[c_idx - 1] + remember[c_idx])
            tmp.append(1)

            # update answer, remember
            answer.append(tmp)
            remember = tmp

        return answer
```

### [119. Pascal's Triangle II](https://leetcode.com/problems/pascals-triangle-ii/)

```python
class Solution:
    def getRow(self, rowIndex: int) -> List[int]:
        if rowIndex == 0:
            return [1]
        elif rowIndex == 1:
            return [1,1]

        t_answer = [1,1] # will be updated during the loop
        for r_idx in range(2, rowIndex + 1):
            # compute new row
            tmp = [1]
            for c_idx in range(1, r_idx):
                tmp.append(t_answer[c_idx - 1] + t_answer[c_idx])
            tmp.append(1)

            # update answer
            t_answer = tmp

        return t_answer
```

class Solution:
    def maxProfit(self, prices: List[int]) -> int:
        min_v = 10 ** 5
        rv = 0
        for price in prices:
            if price - min_v > rv:
                rv = price - min_v
            if price < min_v:
                min_v = price
        return rv

### [338. Counting Bits](https://leetcode.com/problems/counting-bits/)

```python
class Solution:
    def countBits(self, n: int) -> List[int]:
        if n == 0:
            return [0]
        elif n == 1:
            return [0, 1]

        ans = [0]
        for idx in range(1, n + 1):
            tmp = 0
            if idx % 2 == 1:
                tmp = 1
            tmp += ans[idx // 2]
            ans.append(tmp)

        return ans
```

### [392. Is Subsequence](https://leetcode.com/problems/is-subsequence/)

```python
class Solution:
    def isSubsequence(self, s: str, t: str) -> bool:
        if not s:
            return True

        idx_s = 0
        for idx_t in range(len(t)):
            if s[idx_s] == t[idx_t]:
                idx_s += 1
            if idx_s == len(s):
                return True

        return False
```

### [509. Fibonacci Number](https://leetcode.com/problems/fibonacci-number/)

```python
class Solution:
    def fib(self, n: int) -> int:
        if n == 0:
            return 0
        elif n == 1:
            return 1

        prev1 = 0
        prev2 = 1
        for idx in range(2, n + 1):
            tmp = prev1 + prev2
            prev1 = prev2
            prev2 = tmp

        return prev2
```

### [746. Min Cost Climbing Stairs](https://leetcode.com/problems/min-cost-climbing-stairs/)

```python
class Solution:
    def minCostClimbingStairs(self, cost: List[int]) -> int:
        for idx in range(2, len(cost)):
            cost[idx] += min(cost[idx - 1], cost[idx - 2])

        return min(cost[-1], cost[-2])
```

### [1137. N-th Tribonacci Number](https://leetcode.com/problems/n-th-tribonacci-number/)

```python
class Solution:
    def tribonacci(self, n: int) -> int:
        if n == 0:
            return 0
        elif n == 1 or n == 2:
            return 1

        rv = [0] * (n + 1)
        rv[1] = 1
        rv[2] = 1

        for idx in range(3, n + 1):
            rv[idx] = rv[idx - 1] + rv[idx - 2] + rv[idx - 3]

        return rv[-1]
```

### [1668. Maximum Repeating Substring](https://leetcode.com/problems/maximum-repeating-substring/)

```python
class Solution:
    def maxRepeating(self, sequence: str, word: str) -> int:
        n_seq = len(sequence)
        n_word = len(word)

        if n_seq < n_word:
            return 0

        dp = [0] * (n_seq - n_word + 1)
        for idx in range(len(dp)):
            if sequence[idx: idx + n_word] == word:
                dp[idx] = 1

        for idx in range(n_word, len(dp)):
            if dp[idx] == 1:
                dp[idx] += dp[idx - n_word]

        return max(dp)
```

### [2900. Longest Unequal Adjacent Groups Subsequence I](https://leetcode.com/problems/longest-unequal-adjacent-groups-subsequence-i/)

```python
class Solution:
    def getLongestSubsequence(self, words: List[str], groups: List[int]) -> List[str]:
        n = len(words)

        # case a: starts with 0
        prev = 1
        max_a = 0
        for idx in range(n):
            if prev != groups[idx]:
                max_a += 1
                prev = (prev + 1) % 2

        # case b: starts with 1
        prev = 0
        max_b = 0
        for idx in range(n):
            if prev != groups[idx]:
                max_b += 1
                prev = (prev + 1) % 2

        if max_a > max_b:
            prev = 1
        else:
            prev = 0

        rv = []
        for idx in range(n):
            if prev != groups[idx]:
                rv.append(words[idx])
                prev = (prev + 1) % 2

        return rv
```
