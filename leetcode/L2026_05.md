### [LeetCode 70. Climbing Stairs](https://leetcode.com/problems/climbing-stairs/)

```python
class Solution:
    def climbStairs(self, n: int) -> int:
        if n < 2:
            return 1

        tmp1 = tmp2 = 1
        for idx in range(2, n):
            v_idx = tmp1 + tmp2
            tmp1 = tmp2
            tmp2 = v_idx

        return tmp1 + tmp2
```

### [118. Pascal's Triangle](https://leetcode.com/problems/pascals-triangle/)

```python
class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        if numRows == 1:
            return [[1]]
        elif numRows == 2:
            return[[1], [1,1]]

        answer = [[1], [1,1]]
        remember = [1,1]
        for r_idx in range(2, numRows):
            # compute row[r_idx]
            tmp = [1]
            for c_idx in range(1, r_idx):
                tmp.append(remember[c_idx - 1] + remember[c_idx])
            tmp.append(1)

            # update answer, remember
            answer.append(tmp)
            remember = tmp

        return answer
```

### [119. Pascal's Triangle II](https://leetcode.com/problems/pascals-triangle-ii/)

```python
class Solution:
    def getRow(self, rowIndex: int) -> List[int]:
        if rowIndex == 0:
            return [1]
        elif rowIndex == 1:
            return [1,1]

        t_answer = [1,1] # will be updated during the loop
        for r_idx in range(2, rowIndex + 1):
            # compute new row
            tmp = [1]
            for c_idx in range(1, r_idx):
                tmp.append(t_answer[c_idx - 1] + t_answer[c_idx])
            tmp.append(1)

            # update answer
            t_answer = tmp

        return t_answer
```

### [338. Counting Bits](https://leetcode.com/problems/counting-bits/)

```python
class Solution:
    def countBits(self, n: int) -> List[int]:
        if n == 0:
            return [0]
        elif n == 1:
            return [0, 1]

        ans = [0]
        for idx in range(1, n + 1):
            tmp = 0
            if idx % 2 == 1:
                tmp = 1
            tmp += ans[idx // 2]
            ans.append(tmp)

        return ans
```

### [392. Is Subsequence](https://leetcode.com/problems/is-subsequence/)

```python
class Solution:
    def isSubsequence(self, s: str, t: str) -> bool:
        if not s:
            return True

        idx_s = 0
        for idx_t in range(len(t)):
            if s[idx_s] == t[idx_t]:
                idx_s += 1
            if idx_s == len(s):
                return True

        return False
```

class Solution:

        if n == 0:
            return 0
        elif n == 1:
            return 1

        prev2 = 1

            prev1 = prev2
            prev2 = tmp

class Solution:

### [1137. N-th Tribonacci Number](https://leetcode.com/problems/n-th-tribonacci-number/)

```python
class Solution:
    def tribonacci(self, n: int) -> int:
        if n == 0:
            return 0
        elif n == 1 or n == 2:
            return 1

        rv = [0] * (n + 1)
        rv[1] = 1
        rv[2] = 1

        for idx in range(3, n + 1):
            rv[idx] = rv[idx - 1] + rv[idx - 2] + rv[idx - 3]

        return rv[-1]
```

### [1668. Maximum Repeating Substring](https://leetcode.com/problems/maximum-repeating-substring/)

```python
class Solution:
    def maxRepeating(self, sequence: str, word: str) -> int:
        n_seq = len(sequence)
        n_word = len(word)

        if n_seq < n_word:
            return 0

        dp = [0] * (n_seq - n_word + 1)
        for idx in range(len(dp)):
            if sequence[idx: idx + n_word] == word:
                dp[idx] = 1

        for idx in range(n_word, len(dp)):
            if dp[idx] == 1:
                dp[idx] += dp[idx - n_word]

        return max(dp)
```

### [2900. Longest Unequal Adjacent Groups Subsequence I](https://leetcode.com/problems/longest-unequal-adjacent-groups-subsequence-i/)

```python
class Solution:
    def getLongestSubsequence(self, words: List[str], groups: List[int]) -> List[str]:
        n = len(words)

        # case a: starts with 0
        prev = 1
        max_a = 0
        for idx in range(n):
            if prev != groups[idx]:
                max_a += 1
                prev = (prev + 1) % 2

        # case b: starts with 1
        prev = 0
        max_b = 0
        for idx in range(n):
            if prev != groups[idx]:
                max_b += 1
                prev = (prev + 1) % 2

        if max_a > max_b:
            prev = 1
        else:
            prev = 0

        rv = []
        for idx in range(n):
            if prev != groups[idx]:
                rv.append(words[idx])
                prev = (prev + 1) % 2

        return rv
```

### [5. Longest Palindromic Substring](https://leetcode.com/problems/longest-palindromic-substring/)

```python
class Solution:
    def longestPalindrome(self, s: str) -> str:
        n = len(s)
        dp = [[False for _ in range(n + 1)] for _ in range(n + 1)]

        for idx in range(n):
            dp[idx][idx] = True # blank
            dp[idx][idx + 1] = True # single letters
        longest = 1
        rv = s[0]

        for length in range(2, n + 1):
            for idx in range(n):
                if idx + length > n:
                    break
                # length == 5
                # OOOOOX
                # length == 4
                # OOOOX
                if dp[idx + 1][idx + length - 1] and s[idx] == s[idx + length - 1]:
                    dp[idx][idx + length] = True
                    if longest < length:
                        longest = length
                        rv = s[idx: idx + length]

        return rv
```

### [22. Generate Parentheses](https://leetcode.com/problems/generate-parentheses/)

```python
class Solution:
    dp = []

    def compute(idx) -> None:
        for sz1 in range(1, idx):
            for item1 in Solution.dp[sz1]:
                for item2 in Solution.dp[idx - sz1]:
                    Solution.dp[idx].append(item1 + item2)

        for item in Solution.dp[idx - 1]:
            Solution.dp[idx].append("(" + item + ")")

        Solution.dp[idx] = list(set(Solution.dp[idx]))

        return

    def generateParenthesis(self, n: int) -> List[str]:
        if n == 1:
            return ["()"]
        elif n == 2:
            return ["()()", "(())"]

        Solution.dp = [[] for _ in range(n + 1)]
        Solution.dp[1] = ["()"]
        Solution.dp[2] = ["()()", "(())"]
        for idx in range(3, n + 1):
            Solution.compute(idx)

        return Solution.dp[n]
```

### [45. Jump Game II](https://leetcode.com/problems/jump-game-ii/)

```python
class Solution:
    dp = None

    def compute(idx, steps):
        if steps == 0:
            return

        tmp = Solution.dp[idx] + 1
        for u_idx in range(idx + 1, idx + steps + 1):
            if u_idx >= len(Solution.dp):
                return
            if Solution.dp[u_idx] > tmp:
                Solution.dp[u_idx] = tmp

        return

    def jump(self, nums: List[int]) -> int:
        n = len(nums)
        Solution.dp = [10 ** 6] * n
        Solution.dp[0] = 0
        for idx in range(n - 1):
            # check next possible steps
            Solution.compute(idx, nums[idx])

        return Solution.dp[-1]
```

### [53. Maximum Subarray](https://leetcode.com/problems/maximum-subarray/)

```python
class Solution:
    def maxSubArray(self, nums: List[int]) -> int:
        n = len(nums)
        tmp_rv = nums[0]
        rv = nums[0]

        for idx in range(1, n):
            item = nums[idx]
            if tmp_rv + item > item:
                tmp_rv = tmp_rv + item
            else:
                tmp_rv = item

            if rv < tmp_rv:
                rv = tmp_rv

        return rv
```

### [55. Jump Game](https://leetcode.com/problems/jump-game/)

```python
# time limit exceeded
class Solution:
    def canJump(self, nums: List[int]) -> bool:
        n = len(nums)
        r_idx = [0] # collection of reachable index, can be popped and added

        while r_idx:
            for c_idx in r_idx:
                j_lnth = nums[c_idx] # maximum jump length from current index

                # if you cannot jump, continue
                if j_lnth == 0:
                    continue

                # if you can, add all the possible indexes(indices) to r_idx
                for jump in range(1, j_lnth + 1):
                    if c_idx + jump == n - 1:
                        return True # touched the destination
                    elif c_idx + jump not in r_idx and c_idx + jump < n:
                        r_idx.append(c_idx + jump)

                # remove the computed index from r_idx
                r_idx.remove(c_idx)

        return False
```

        for c_idx, jmp in enumerate(nums):
            if c_idx > rng:

            # update rng
            rng = max(rng, c_idx + jmp)

            if rng >= n - 1:
