### [LeetCode 70. Climbing Stairs](https://leetcode.com/problems/climbing-stairs/)

```python
class Solution:
    def climbStairs(self, n: int) -> int:
        if n < 2:
            return 1

        tmp1 = tmp2 = 1
        for idx in range(2, n):
            v_idx = tmp1 + tmp2
            tmp1 = tmp2
            tmp2 = v_idx

        return tmp1 + tmp2
```

### [118. Pascal's Triangle](https://leetcode.com/problems/pascals-triangle/)

```python
class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        if numRows == 1:
            return [[1]]
        elif numRows == 2:
            return[[1], [1,1]]

        answer = [[1], [1,1]]
        remember = [1,1]
        for r_idx in range(2, numRows):
            # compute row[r_idx]
            tmp = [1]
            for c_idx in range(1, r_idx):
                tmp.append(remember[c_idx - 1] + remember[c_idx])
            tmp.append(1)

            # update answer, remember
            answer.append(tmp)
            remember = tmp

        return answer
```

```python
            # update answer
            t_answer = tmp

        return t_answer
```

```python
        if n == 0:
            return [0]
        elif n == 1:
            return [0, 1]

        ans = [0]
        for idx in range(1, n + 1):

            tmp += ans[idx // 2]
            ans.append(tmp)
```

```python
class Solution:

        if not s:
            return True

        idx_s = 0
        for idx_t in range(len(t)):

                idx_s += 1
            if idx_s == len(s):
```

```python
class Solution:
    def tribonacci(self, n: int) -> int:
        if n == 0:
            return 0
        elif n == 1 or n == 2:
            return 1

        rv = [0] * (n + 1)
        rv[1] = 1
        rv[2] = 1

        for idx in range(3, n + 1):
            rv[idx] = rv[idx - 1] + rv[idx - 2] + rv[idx - 3]

        return rv[-1]
```

### [1668. Maximum Repeating Substring](https://leetcode.com/problems/maximum-repeating-substring/)

```python
class Solution:
    def maxRepeating(self, sequence: str, word: str) -> int:
        n_seq = len(sequence)
        n_word = len(word)

        if n_seq < n_word:
            return 0

        dp = [0] * (n_seq - n_word + 1)
        for idx in range(len(dp)):
            if sequence[idx: idx + n_word] == word:
                dp[idx] = 1

        for idx in range(n_word, len(dp)):
            if dp[idx] == 1:
                dp[idx] += dp[idx - n_word]

        return max(dp)
```

### [2900. Longest Unequal Adjacent Groups Subsequence I](https://leetcode.com/problems/longest-unequal-adjacent-groups-subsequence-i/)

```python
class Solution:
    def getLongestSubsequence(self, words: List[str], groups: List[int]) -> List[str]:
        n = len(words)

        # case a: starts with 0
        prev = 1
        max_a = 0
        for idx in range(n):
            if prev != groups[idx]:
                max_a += 1
                prev = (prev + 1) % 2

        # case b: starts with 1
        prev = 0
        max_b = 0
        for idx in range(n):
            if prev != groups[idx]:
                max_b += 1
                prev = (prev + 1) % 2

        if max_a > max_b:
            prev = 1
        else:
            prev = 0

        rv = []
        for idx in range(n):
            if prev != groups[idx]:
                rv.append(words[idx])
                prev = (prev + 1) % 2

        return rv
```

### [5. Longest Palindromic Substring](https://leetcode.com/problems/longest-palindromic-substring/)

```python
class Solution:
    def longestPalindrome(self, s: str) -> str:
        n = len(s)
        dp = [[False for _ in range(n + 1)] for _ in range(n + 1)]

        for idx in range(n):
            dp[idx][idx] = True # blank
            dp[idx][idx + 1] = True # single letters
        longest = 1
        rv = s[0]

        for length in range(2, n + 1):
            for idx in range(n):
                if idx + length > n:
                    break
                # length == 5
                # OOOOOX
                # length == 4
                # OOOOX
                if dp[idx + 1][idx + length - 1] and s[idx] == s[idx + length - 1]:
                    dp[idx][idx + length] = True
                    if longest < length:
                        longest = length
                        rv = s[idx: idx + length]

        return rv
```

### [45. Jump Game II](https://leetcode.com/problems/jump-game-ii/)

```python
class Solution:
    dp = None

    def compute(idx, steps):
        if steps == 0:
            return

        tmp = Solution.dp[idx] + 1
        for u_idx in range(idx + 1, idx + steps + 1):
            if u_idx >= len(Solution.dp):
                return
            if Solution.dp[u_idx] > tmp:
                Solution.dp[u_idx] = tmp

        return

    def jump(self, nums: List[int]) -> int:
        n = len(nums)
        Solution.dp = [10 ** 6] * n
        Solution.dp[0] = 0
        for idx in range(n - 1):
            # check next possible steps
            Solution.compute(idx, nums[idx])

        return Solution.dp[-1]
```

```python
class Solution:
    def canJump(self, nums: List[int]) -> bool:
        n = len(nums)
        r_idx = [0] # collection of reachable index, can be popped and added

        while r_idx:
            for c_idx in r_idx:
                j_lnth = nums[c_idx] # maximum jump length from current index

                # if you cannot jump, continue
                if j_lnth == 0:
                    continue

                # if you can, add all the possible indexes(indices) to r_idx
                for jump in range(1, j_lnth + 1):
                    if c_idx + jump == n - 1:
                        return True # touched the destination
                    elif c_idx + jump not in r_idx and c_idx + jump < n:
                        r_idx.append(c_idx + jump)

                # remove the computed index from r_idx
                r_idx.remove(c_idx)

        return False
```

### [62. Unique Paths](https://leetcode.com/problems/unique-paths/)

```python
class Solution:
    def uniquePaths(self, m: int, n: int) -> int:
        dp = [[-1 for _ in range(n)] for _ in range(m)]
        for idx in range(m):
            dp[idx][0] = 1
        for idx in range(n):
            dp[0][idx] = 1

        for r_idx in range(1, m):
            for c_idx in range(1, n):
                dp[r_idx][c_idx] = dp[r_idx - 1][c_idx] + dp[r_idx][c_idx - 1]
        
        return dp[m - 1][n - 1]
```

### [63. Unique Paths II](https://leetcode.com/problems/unique-paths-ii/)

```python
class Solution:
    def uniquePathsWithObstacles(self, obstacleGrid: List[List[int]]) -> int:
        m = len(obstacleGrid)
        n = len(obstacleGrid[0])
        dp = [[0 for _ in range(n)] for _ in range(m)]
        
        # set up
        for idx in range(m):
            if obstacleGrid[idx][0] == 1:
                break
            dp[idx][0] = 1
        for idx in range(n):
            if obstacleGrid[0][idx] == 1:
                break
            dp[0][idx] = 1
        
        # compute
        for r in range(1, m):
            for c in range(1, n):
```

### [64. Minimum Path Sum](https://leetcode.com/problems/minimum-path-sum/)

```python
class Solution:
    def minPathSum(self, grid: List[List[int]]) -> int:
        m = len(grid)
        n = len(grid[0])

        dp = [[0 for _ in range(n)] for _ in range(m)]
        dp[0][0] = grid[0][0]

        for r_idx in range(1, m):
            dp[r_idx][0] = dp[r_idx - 1][0] + grid[r_idx][0]
        for c_idx in range(1, n):
            dp[0][c_idx] = dp[0][c_idx - 1] + grid[0][c_idx]
        
        for r_idx in range(1, m):
            for c_idx in range(1, n):
                dp[r_idx][c_idx] = min(dp[r_idx - 1][c_idx], dp[r_idx][c_idx - 1]) + grid[r_idx][c_idx]
        
        return dp[m - 1][n - 1]
```

### [72. Edit Distance](https://leetcode.com/problems/edit-distance/)

```python
class Solution:
    def minDistance(self, word1: str, word2: str) -> int:
        m = len(word1)
        n = len(word2)
        
        dp = [[0 for _ in range(n + 1)] for _ in range(m + 1)]

        for idx in range(1, m + 1):
            dp[idx][0] = idx
        for idx in range(1, n + 1):
            dp[0][idx] = idx
        
        for r_idx in range(1, m + 1):
            for c_idx in range(1, n + 1):
                if word1[r_idx - 1] == word2[c_idx - 1]:
                    dp[r_idx][c_idx] = dp[r_idx - 1][c_idx - 1]
                else:
                    cand1 = dp[r_idx - 1][c_idx] + 1
                    cand2 = dp[r_idx][c_idx - 1] + 1
                    cand3 = dp[r_idx - 1][c_idx - 1] + 1
                    dp[r_idx][c_idx] = min(cand1, cand2, cand3)
        
        return dp[m][n]
```

### [91. Decode Ways](https://leetcode.com/problems/decode-ways/)

```python
class Solution:
    def isPossible(s: str):
        n = len(s)
        tmp = int(s)
        
        if n == 1 and (1 <= tmp <= 9):
            return True
        elif n == 2 and (10 <= tmp <= 26):
            return True

        return False

    def isDivergent(s: str):
        tmp = int(s)
        
        if 10 < tmp < 20 or 20 < tmp <= 26:
            return True
        
        return False

    def numDecodings(self, s: str) -> int:
        n = len(s)
        dp = [0 for _ in range(n + 1)]

        if int(s[0]) == 0:
            return 0
        dp[0] = dp[1] = 1
        
        for idx in range(2, n + 1):
            # adding one letter of one digit
            if dp[idx - 1] != 0 and Solution.isPossible(s[idx - 1: idx]):
                dp[idx] += dp[idx - 1]

            # adding one letter of two digits
            if dp[idx - 2] != 0 and Solution.isPossible(s[idx - 2: idx]):
                dp[idx] += dp[idx - 2]
        
        return dp[n]
```

### [95. Unique Binary Search Trees II](https://leetcode.com/problems/unique-binary-search-trees-ii/)

```python
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right

class Solution:
    mztn = {}

    def compute(idx1, idx2) -> List[Optional[TreeNode]]:
        """
        returns list of all the possible trees within the given range(idx1, idx2)
        """
        if (idx1, idx2) in Solution.mztn:
            return Solution.mztn[(idx1, idx2)]

        rv = []
        for pivot in range(idx1, idx2):
            left_children = []
            if idx1 < pivot:
                left_children = Solution.compute(idx1, pivot)
            else:
                left_children = [None]

            right_children = []
            if pivot + 1 < idx2:
                right_children = Solution.compute(pivot + 1, idx2)
            else:
                right_children = [None]
            
            for left in left_children:
                for right in right_children:
                    tmp = TreeNode(pivot)
                    tmp.left = left
                    tmp.right = right
                    rv.append(tmp)
        
        Solution.mztn[(idx1, idx2)] = rv
        return rv

    def generateTrees(self, n: int) -> List[Optional[TreeNode]]:
        for idx in range(1, n + 1):
            Solution.mztn[(idx, idx + 1)] = [TreeNode(idx)]
        return Solution.compute(1, n + 1)
```
