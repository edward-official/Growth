### [LeetCode 70. Climbing Stairs](https://leetcode.com/problems/climbing-stairs/)

```python
class Solution:
    def climbStairs(self, n: int) -> int:
        if n < 2:
            return 1

        tmp1 = tmp2 = 1
        for idx in range(2, n):
            v_idx = tmp1 + tmp2
            tmp1 = tmp2
            tmp2 = v_idx

        return tmp1 + tmp2
```

### [118. Pascal's Triangle](https://leetcode.com/problems/pascals-triangle/)

```python
class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        if numRows == 1:
            return [[1]]
        elif numRows == 2:
            return[[1], [1,1]]

        answer = [[1], [1,1]]
        remember = [1,1]
        for r_idx in range(2, numRows):
            # compute row[r_idx]
            tmp = [1]
            for c_idx in range(1, r_idx):
                tmp.append(remember[c_idx - 1] + remember[c_idx])
            tmp.append(1)

            # update answer, remember
            answer.append(tmp)
            remember = tmp

        return answer
```

### [119. Pascal's Triangle II](https://leetcode.com/problems/pascals-triangle-ii/)

```python
class Solution:
    def getRow(self, rowIndex: int) -> List[int]:
        if rowIndex == 0:
            return [1]
        elif rowIndex == 1:
            return [1,1]

        t_answer = [1,1] # will be updated during the loop
        for r_idx in range(2, rowIndex + 1):
            # compute new row
            tmp = [1]
            for c_idx in range(1, r_idx):
                tmp.append(t_answer[c_idx - 1] + t_answer[c_idx])
            tmp.append(1)

            # update answer
            t_answer = tmp

        return t_answer
```
