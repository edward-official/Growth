### [409. Longest Palindrome](https://leetcode.com/problems/longest-palindrome/)

```python
class Solution:
    def longestPalindrome(self, s: str) -> int:
        isCenterOdd = False
        cnt_map = {}
        rv = 0

        for ch in s:
            if ch not in cnt_map:
                cnt_map[ch] = 1
            else:
                cnt_map[ch] += 1

        for v in cnt_map.values():
            if v % 2 != 0:
                isCenterOdd = True
            rv += (v // 2) * 2

        return rv + (1 if isCenterOdd else 0)
```

### [455. Assign Cookies](https://leetcode.com/problems/assign-cookies/)

```python
class Solution:
    def findContentChildren(self, g: List[int], s: List[int]) -> int:
        rv = 0
        g.sort(reverse = True)
        s.sort(reverse = True)

        for grdy in g:
            # terminate when there's no cookie left
            if not s:
                break

            # hand over the cookie when the size is enough
            if grdy <= s[0]:
                rv += 1
                s.pop(0)

        return rv
```

### [561. Array Partition](https://leetcode.com/problems/array-partition/)

```python
class Solution:
    def arrayPairSum(self, nums: List[int]) -> int:
        n = len(nums)
        nums.sort()
        rv = 0

        for idx in range(n):
            if idx % 2 == 0:
                rv += nums[idx]

        return rv
```

### [605. Can Place Flowers](https://leetcode.com/problems/can-place-flowers/)

```python
class Solution:
    def canPlaceFlowers(self, flowerbed: List[int], n: int) -> bool:
        sz = len(flowerbed)

        if sz == 1 and flowerbed[0] == 0:
            if n <= 1:
                return True
            return False
        elif sz == 1 and flowerbed[0] != 0:
            if n == 0:
                return True
            return False

        cnt = 0
        if sz >= 2 and flowerbed[0] == 0 and flowerbed[1] == 0:
            flowerbed[0] = 1
            cnt += 1

        for idx in range(1, sz - 1):
            if flowerbed[idx - 1] == 0 and flowerbed[idx] == 0 and flowerbed[idx + 1] == 0:
                flowerbed[idx] = 1
                cnt += 1

        if sz >= 2 and flowerbed[sz - 2] == 0 and flowerbed[sz - 1] == 0:
            flowerbed[sz - 1] = 1
            cnt += 1

        if cnt < n:
            return False
        return True
```

```python
class Solution:
    def validPalindrome(self, s: str) -> bool:
        idx1 = 0
        idx2 = len(s) - 1

        # proceed until situation to delete
        while idx1 < idx2:
            if s[idx1] != s[idx2]:
                break
            idx1 += 1
            idx2 -= 1

        def isPalindrome(s, idx1, idx2):
            while idx1 < idx2:
                if s[idx1] != s[idx2]:
                    return False
                idx1 += 1
                idx2 -= 1

            return True

        if isPalindrome(s, idx1 + 1, idx2) or isPalindrome(s, idx1, idx2 - 1):
            return True
        return False
```
