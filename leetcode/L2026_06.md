### [409. Longest Palindrome](https://leetcode.com/problems/longest-palindrome/)

```python
class Solution:
    def longestPalindrome(self, s: str) -> int:
        isCenterOdd = False
        cnt_map = {}
        rv = 0

        for ch in s:
            if ch not in cnt_map:
                cnt_map[ch] = 1
            else:
                cnt_map[ch] += 1

        for v in cnt_map.values():
            if v % 2 != 0:
                isCenterOdd = True
            rv += (v // 2) * 2

        return rv + (1 if isCenterOdd else 0)
```

```python
class Solution:
    def findContentChildren(self, g: List[int], s: List[int]) -> int:
        rv = 0
        g.sort(reverse = True)
        s.sort(reverse = True)

        for grdy in g:
            # terminate when there's no cookie left
```

```python
class Solution:
    def canPlaceFlowers(self, flowerbed: List[int], n: int) -> bool:
        sz = len(flowerbed)

        if sz == 1 and flowerbed[0] == 0:
            if n <= 1:
                return True
            return False
        elif sz == 1 and flowerbed[0] != 0:
            if n == 0:
                return True
            return False

        cnt = 0
        if sz >= 2 and flowerbed[0] == 0 and flowerbed[1] == 0:
            flowerbed[0] = 1
            cnt += 1

        for idx in range(1, sz - 1):
            if flowerbed[idx - 1] == 0 and flowerbed[idx] == 0 and flowerbed[idx + 1] == 0:
                flowerbed[idx] = 1
                cnt += 1

        if sz >= 2 and flowerbed[sz - 2] == 0 and flowerbed[sz - 1] == 0:
            flowerbed[sz - 1] = 1
            cnt += 1

        if cnt < n:
            return False
        return True
```

### [942. DI String Match](https://leetcode.com/problems/di-string-match/)

```python
class Solution:
    def diStringMatch(self, s: str) -> List[int]:
        """
        IDID
        I: [0] -> [0, 1]
        D: [0, 1] -> [0, 1, -1] -> [1, 2, 0]
        I: [1, 2, 0] -> [1, 2, 0, 3]
        D: [1, 2, 0, 3] -> [1, 2, 0, 3, -1] -> [2, 3, 1, 4, 0]
        """

        def addOne(lst):
            for idx in range(len(lst)):
                lst[idx] += 1

        rv = [0]
        for t in s:
            if t == "I":
                rv.append(len(rv))
            else:
                rv.append(-1)
                addOne(rv)

        return rv
```

### [1005. Maximize Sum Of Array After K Negations](https://leetcode.com/problems/maximize-sum-of-array-after-k-negations/)

```python
class Solution:
    def largestSumAfterKNegations(self, nums: List[int], k: int) -> int:
        n = len(nums)

        # first flip as many the negative numbers as possible
        nums.sort()
        for idx in range(n):
            if k == 0:
                break

            if nums[idx] < 0:
                nums[idx] = (-1) * nums[idx]
                k -= 1

        if k % 2 == 0:
            return sum(nums)

        nums.sort()
        nums[0] = (-1) * nums[0]
        return sum(nums)
```

```python
class Solution:
    def canThreePartsEqualSum(self, arr: List[int]) -> bool:
        n = len(arr)
        ts = sum(arr) # target sum
        if ts % 3 != 0:
            return False

        ts = ts // 3
        tmp_sum = 0
        cnt = 0
        rdx = None # remember idx
        for idx in range(n):
            tmp_sum += arr[idx]
            if tmp_sum == ts:
                cnt += 1
                tmp_sum = 0
                if cnt == 2:
                    rdx = idx + 1
                    break

            return False

        if rdx <= n - 1 and sum(arr[rdx:]) == ts:
            return True
        return False
```

```python
class Solution:
```

```python
        boxTypes.sort(key = lambda x: x[1], reverse = True)

        rv = 0
        remain = truckSize # space left for further boxes
        for b, u in boxTypes:
            rv += min(remain, b) * u
            remain -= min(remain, b)
            if remain == 0:
                break

        return rv
```

```python
        n = len(nums)
        rv = 0
        for idx in range(1, n):

                nums[idx] += tmp

        return rv
```
