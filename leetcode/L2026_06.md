### [409. Longest Palindrome](https://leetcode.com/problems/longest-palindrome/)

```python
class Solution:
    def longestPalindrome(self, s: str) -> int:
        isCenterOdd = False
        cnt_map = {}
        rv = 0

        for ch in s:
            if ch not in cnt_map:
                cnt_map[ch] = 1
            else:
                cnt_map[ch] += 1

        for v in cnt_map.values():
            if v % 2 != 0:
                isCenterOdd = True
            rv += (v // 2) * 2

        return rv + (1 if isCenterOdd else 0)
```

### [455. Assign Cookies](https://leetcode.com/problems/assign-cookies/)

```python
class Solution:
    def findContentChildren(self, g: List[int], s: List[int]) -> int:
        rv = 0
        g.sort(reverse = True)
        s.sort(reverse = True)

        for grdy in g:
            # terminate when there's no cookie left
            if not s:
                break

            # hand over the cookie when the size is enough
            if grdy <= s[0]:
                rv += 1
                s.pop(0)

        return rv
```

### [605. Can Place Flowers](https://leetcode.com/problems/can-place-flowers/)

```python
class Solution:
    def canPlaceFlowers(self, flowerbed: List[int], n: int) -> bool:
        sz = len(flowerbed)

        if sz == 1 and flowerbed[0] == 0:
            if n <= 1:
                return True
            return False
        elif sz == 1 and flowerbed[0] != 0:
            if n == 0:
                return True
            return False

        cnt = 0
        if sz >= 2 and flowerbed[0] == 0 and flowerbed[1] == 0:
            flowerbed[0] = 1
            cnt += 1

        for idx in range(1, sz - 1):
            if flowerbed[idx - 1] == 0 and flowerbed[idx] == 0 and flowerbed[idx + 1] == 0:
                flowerbed[idx] = 1
                cnt += 1

        if sz >= 2 and flowerbed[sz - 2] == 0 and flowerbed[sz - 1] == 0:
            flowerbed[sz - 1] = 1
            cnt += 1

        if cnt < n:
            return False
        return True
```

class Solution:
    def validPalindrome(self, s: str) -> bool:
        idx1 = 0
        idx2 = len(s) - 1

        # proceed until situation to delete
        while idx1 < idx2:
            if s[idx1] != s[idx2]:
                break
            idx1 += 1
            idx2 -= 1

        def isPalindrome(s, idx1, idx2):
            while idx1 < idx2:
                if s[idx1] != s[idx2]:
                    return False
                idx1 += 1
                idx2 -= 1

            return True

        if isPalindrome(s, idx1 + 1, idx2) or isPalindrome(s, idx1, idx2 - 1):
            return True
        return False

### [860. Lemonade Change](https://leetcode.com/problems/lemonade-change/)

```python
class Solution:
    def lemonadeChange(self, bills: List[int]) -> bool:
        billcount = [0, 0]

        for b in bills:
            if b == 5:
                billcount[0] += 1
            elif b == 10:
                if billcount[0] > 0:
                    billcount[0] -= 1
                else:
                    return False

                billcount[1] += 1
            elif b == 20:
                if billcount[1] > 0 and billcount[0] > 0:
                    billcount[1] -= 1
                    billcount[0] -= 1
                elif billcount[0] >= 3:
                    billcount[0] -= 3
                else:
                    return False

        return True
```

### [942. DI String Match](https://leetcode.com/problems/di-string-match/)

```python
class Solution:
    def diStringMatch(self, s: str) -> List[int]:
        """
        IDID
        I: [0] -> [0, 1]
        D: [0, 1] -> [0, 1, -1] -> [1, 2, 0]
        I: [1, 2, 0] -> [1, 2, 0, 3]
        D: [1, 2, 0, 3] -> [1, 2, 0, 3, -1] -> [2, 3, 1, 4, 0]
        """

        def addOne(lst):
            for idx in range(len(lst)):
                lst[idx] += 1

        rv = [0]
        for t in s:
            if t == "I":
                rv.append(len(rv))
            else:
                rv.append(-1)
                addOne(rv)

        return rv
```

### [976. Largest Perimeter Triangle](https://leetcode.com/problems/largest-perimeter-triangle/)

```python
class Solution:
    def largestPerimeter(self, nums: List[int]) -> int:
        nums.sort(reverse = True)
        n = len(nums)

        for idx in range(n - 2):
            if nums[idx] < nums[idx + 1] + nums[idx + 2]:
                return nums[idx] + nums[idx + 1] + nums[idx + 2]

        return 0
```

### [1005. Maximize Sum Of Array After K Negations](https://leetcode.com/problems/maximize-sum-of-array-after-k-negations/)

```python
class Solution:
    def largestSumAfterKNegations(self, nums: List[int], k: int) -> int:
        n = len(nums)

        # first flip as many the negative numbers as possible
        nums.sort()
        for idx in range(n):
            if k == 0:
                break

            if nums[idx] < 0:
                nums[idx] = (-1) * nums[idx]
                k -= 1

        if k % 2 == 0:
            return sum(nums)

        nums.sort()
        nums[0] = (-1) * nums[0]
        return sum(nums)
```

### [1013. Partition Array Into Three Parts With Equal Sum](https://leetcode.com/problems/partition-array-into-three-parts-with-equal-sum/)

```python
class Solution:
    def canThreePartsEqualSum(self, arr: List[int]) -> bool:
        n = len(arr)
        ts = sum(arr) # target sum
        if ts % 3 != 0:
            return False

        ts = ts // 3
        tmp_sum = 0
        cnt = 0
        rdx = None # remember idx
        for idx in range(n):
            tmp_sum += arr[idx]
            if tmp_sum == ts:
                cnt += 1
                tmp_sum = 0
                if cnt == 2:
                    rdx = idx + 1
                    break

        if not rdx:
            return False

        if rdx <= n - 1 and sum(arr[rdx:]) == ts:
            return True
        return False
```

### [1217. Minimum Cost to Move Chips to The Same Position](https://leetcode.com/problems/minimum-cost-to-move-chips-to-the-same-position/)

```python
class Solution:
    def minCostToMoveChips(self, position: List[int]) -> int:
        cnt = {}
        for p in position:
            if p not in cnt:
                cnt[p] = 1
            else:
                cnt[p] += 1

        esum = osum = 0
        for p in cnt:
            if p % 2 == 0:
                esum += cnt[p]
            else:
                osum += cnt[p]

        return min(esum, osum)
```

### [1221. Split a String in Balanced Strings](https://leetcode.com/problems/split-a-string-in-balanced-strings/)

```python
class Solution:
    def balancedStringSplit(self, s: str) -> int:
        rv = 0
        Lcnt = Rcnt = 0

        for c in s:
            if c == "L":
                Lcnt += 1
            elif c == "R":
                Rcnt += 1

            if Lcnt == Rcnt:
                rv += 1
                Lcnt = Rcnt = 0

        return rv
```

### [1403. Minimum Subsequence in Non-Increasing Order](https://leetcode.com/problems/minimum-subsequence-in-non-increasing-order/)

```python
class Solution:
    def minSubsequence(self, nums: List[int]) -> List[int]:
        half = sum(nums) // 2
        n = len(nums)

        nums.sort(reverse=True)
        tmp = 0
        rv = []
        for v in nums:
            tmp += v
            rv.append(v)
            if half < tmp:
                break

        return sorted(rv, reverse=True)
```
