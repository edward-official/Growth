### 1. [교점에 별 만들기](https://programmers.co.kr/learn/courses/30/lessons/87377)

```python
def solution(line):
    def get_point(line1, line2):
        a, b, e = line1
        c, d, f = line2
        if a * d - b * c == 0:
            return None
        elif (b * f - e * d) % (a * d - b * c) != 0:
            return None
        elif (e * c - a * f) % (a * d - b * c) != 0:
            return None
        else:
            return ((b * f - e * d) // (a * d - b * c), (e * c - a * f) // (a * d - b * c))

    points = []
    for i in range(len(line)):
        for j in range(i + 1, len(line)):
            got = get_point(line[i], line[j])
            if got:
                points.append(got)

    x_max = max([p[0] for p in points])
    x_min = min([p[0] for p in points])
    y_max = max([p[1] for p in points])
    y_min = min([p[1] for p in points])

    answer = [["." for x in range(x_max - x_min + 1)] for y in range(y_max - y_min + 1)]
    for x, y in points:
        answer[y_max - y][x - x_min] = "*"

    return ["".join(_rows) for _rows in answer]
```

### 2. [행렬 테두리 회전하기](https://programmers.co.kr/learn/courses/30/lessons/77485)

```python
def solution(rows, columns, queries):
    def rotate(table, r_info):
        r1, c1, r2, c2 = [t - 1 for t in r_info]
        group_up = table[r1][c1: c2]
        group_down = table[r2][c1 + 1: c2 + 1]
        rv = min(min(group_up), min(group_down))

        for r in range(r2, r1, -1):
            table[r][c2] = table[r - 1][c2]
            if table[r][c2] < rv:
                rv = table[r][c2]
        for r in range(r1, r2):
            table[r][c1] = table[r + 1][c1]
            if table[r][c1] < rv:
                rv = table[r][c1]

        table[r1][c1 + 1: c2 + 1] = group_up
        table[r2][c1: c2] = group_down

        return rv

    answer = []
    table = [[(r * columns + c + 1) for c in range(columns)] for r in range(rows)]
    for q in queries:
        answer.append(rotate(table, q))

    return answer
```

### 3. [삼각 달팽이](https://programmers.co.kr/learn/courses/30/lessons/68645)

```python
def oneline(snail, direction, r, c, n_element, v_start):
    dr, dc = direction
    for t in range(n_element):
        snail[r + dr * t][c + dc * t] = v_start + t

    return (r + dr * (n_element - 1), c + dc * (n_element - 1), v_start + n_element)

def solution(n):
    snail = [[-1 for j in range(i)] for i in range(1, n + 1)]
    d_choice = 0
    d_options = [(1, 0), (0, 1), (-1, -1)]
    r_start = -1
    c_start = 0
    v_start = 1
    for n_element in range(n, 0, -1):
        r_fix, c_fix = d_options[d_choice]
        r_start, c_start, v_start = oneline(snail, d_options[d_choice], r_start + r_fix, c_start + c_fix, n_element, v_start)
        d_choice = (d_choice + 1) % 3

    answer = []
    for r in snail:
        answer += r
    return answer
```

### 4. [거리두기 확인하기](https://programmers.co.kr/learn/courses/30/lessons/81302)

```python
def check_room(place):
    for r in range(5):
        for c in range(5):
            if place[r][c] != "P":
                continue
            if (r + 1) < 5 and place[r + 1][c] == "P":
                return 0
            elif (c + 1) < 5 and place[r][c + 1] == "P":
                return 0
            elif (r + 2) < 5 and place[r + 2][c] == "P" and place[r + 1][c] != "X":
                return 0
            elif (r + 1) < 5 and (c + 1) < 5 and place[r + 1][c + 1] == "P" and (place[r + 1][c] != "X" or place[r][c + 1] != "X"):
                return 0
            elif (r - 1) >= 0 and (c + 1) < 5 and place[r - 1][c + 1] == "P" and (place[r - 1][c] != "X" or place[r][c + 1] != "X"):
                return 0
            elif (r + 1) < 5 and (c - 1) >= 0 and place[r + 1][c - 1] == "P" and (place[r + 1][c] != "X" or place[r][c - 1] != "X"):
                return 0
            elif (c + 2) < 5 and place[r][c + 2] == "P" and place[r][c + 1] != "X":
                return 0
    return 1

def solution(places):
    answer = []
    for p in places:
        answer.append(check_room(p))
    return answer
```
