## [문제 8: 튜플](https://school.programmers.co.kr/learn/courses/30/lessons/64065)

### 나의 풀이

```python
def solution(s):
    divide1 = s[2:-2].split("},{")
    divide1.sort(key = lambda x: len(x))
    t_hash = {}

    for element1 in divide1:
        divide2 = element1.split(",")
        for element2 in divide2:
            if element2 not in t_hash:
                t_hash[int(element2)] = True

    return list(t_hash)
```

```python
def solution(s):
    letter_stack = []
    for letter in s:
        if letter_stack and letter == letter_stack[-1]:
            letter_stack.pop()
        else:
            letter_stack.append(letter)

    return 0 if letter_stack else 1
```

## [문제 10: 문자열 압축](https://programmers.co.kr/learn/courses/30/lessons/60057)

```python
def compute_length(item, length):
    rv = 0

    previous = None
    current = item[0: length]
    count = 1

    for idx in range(length, len(item), length):
        previous = current
        current = item[idx: idx + length]
        if previous == current:
            count += 1
        else:
            if count == 1:
                rv += length
            else:
                rv += (length + len(str(count)))
            count = 1

    if count == 1:
        rv += len(current)
    else:
        rv += (len(current) + len(str(count)))

    return rv

def solution(item):
    rv = len(item) # How to use infinite value?

    for length in range(1, len(item) // 2 + 1):
        temp = compute_length(item, length)
        if temp < rv:
            rv = temp

    return rv
```

## [문제 11: 3진법 뒤집기](https://programmers.co.kr/learn/courses/30/lessons/68935)

### 나의 풀이

```python
def radix(item, rdx):
    if item == 0:
        return "0"

    array = []
    while item:
        t = item % rdx
        item = item // rdx

        if t < 10:
            array.append(str(t))
        else:
            array.append(chr(ord("A") + t - 10))

    return "".join(reversed(array))

def solution(n):
    convert1 = radix(n, 3)
    return int("".join(reversed(convert1)), 3)
```

## [문제 12: 이진 변환 반복하기](https://programmers.co.kr/learn/courses/30/lessons/70129)

### 나의 풀이

```python
def process(s):
    n_deleted = 0
    n_survive = 0
    
    for letter in s:
        if letter == "0":
            n_deleted += 1
        elif letter == "1":
            n_survive += 1
    
    return (n_deleted, n_survive)

def solution(s):
    n_process = 0
    n_deleted = 0
    
    while s != "1":
        t_deleted, t_survive = process(s)
        n_process += 1
        n_deleted += t_deleted
        s = bin(t_survive)[2:]
    
    return [n_process, n_deleted]
```

## [문제 13: 신규 아이디 추천](https://school.programmers.co.kr/learn/courses/30/lessons/72410)

### 나의 풀이

```python
def solution(new_id):
    new_id = step1(new_id)
    new_id = step2(new_id)
    new_id = step3(new_id)
    new_id = step4(new_id)
    new_id = step5(new_id)
    new_id = step6(new_id)
    new_id = step7(new_id)
    return new_id

def step1(target):
    return target.lower()

def step2(target):
    temp = []
    for letter in target:
        if letter.isdigit() or letter.isalpha() or letter in ["-", "_", "."]:
            temp.append(letter)
    return "".join(temp)

def step3(target):
    while ".." in target:
        target = target.replace("..", ".")
    return target

def step4(target):
    return target.strip(".")

def step5(target):
    if not target:
        return "a"
    return target

def step6(target):
    if len(target) >= 16:
        return target[:15] if target[14] != "." else target[:14]
    return target

def step7(target):
    if len(target) <= 2:
        while len(target) != 3:
            target += target[-1]
    return target
```
